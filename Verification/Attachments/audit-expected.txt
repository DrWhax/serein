def Attachments.shapeOK : Nat → Bool :=
fun len => decide (ivLength + 16 + macLength ≤ len) && (len - ivLength - macLength) % 16 == 0
def Attachments.decrypt : Blob → Except Err Nat :=
fun b =>
  if (!b.keyOK) = true then Except.error Err.badKey
  else
    if (!shapeOK b.len) = true then Except.error Err.tooShort
    else
      if (!b.macOK) = true then Except.error Err.badMAC
      else
        if (b.inc == Inc.bad || b.inc == Inc.huge) = true then Except.error Err.badIncrementalMAC
        else
          match b.plain with
          | none => Except.error Err.badPadding
          | some p =>
            have n := min b.size p;
            match b.integrity with
            | Integrity.digest empty good =>
              if empty = true then Except.error Err.missingDigest
              else if (!good) = true then Except.error Err.badDigest else Except.ok n
            | Integrity.hash good => if (!good) = true then Except.error Err.badPlaintextHash else Except.ok n
            | Integrity.macOnly => Except.ok n
def Attachments.ciphertextSize : Nat → Nat :=
fun pad => ivLength + (1 + pad / 16) * 16 + macLength

def Attachments.describe : Att → Resp → Blob :=
fun a r =>
  have own := r.genuine == some a.upload;
  { keyOK := true, len := r.len, macOK := own, inc := Inc.none,
    integrity :=
      if a.hasDigest = true then Integrity.digest false own
      else if a.hasHash = true then Integrity.hash (own && a.size == a.plainLen) else Integrity.digest true false,
    size := a.size, plain := some a.pad }
def Attachments.setAtt : List Att → Nat → (Att → Att) → List Att :=
fun atts id f => List.map (fun x => if (x.id == id) = true then f x else x) atts
def Attachments.download : S → Nat → Resp → S :=
fun s id r =>
  match List.find? (fun x => x.id == id) s.atts with
  | none => s
  | some a =>
    if (a.st != St.pending) = true then s
    else
      if s.maxSize < a.size then
        {
          atts :=
            setAtt s.atts id fun x =>
              { id := x.id, upload := x.upload, size := x.size, plainLen := x.plainLen, pad := x.pad,
                bucket := x.bucket, hasDigest := x.hasDigest, hasHash := x.hasHash, st := St.tooBig, path := x.path },
          files := s.files, nextPath := s.nextPath, maxSize := s.maxSize, delivered := s.delivered,
          fetched := s.fetched }
      else
        have cap := ciphertextSize a.bucket + margin;
        have held := min r.len (cap + 1);
        have s :=
          { atts := s.atts, files := s.files, nextPath := s.nextPath, maxSize := s.maxSize, delivered := s.delivered,
            fetched := s.fetched ++ [(id, held)] };
        if cap < r.len then s
        else
          match decrypt (describe a r) with
          | Except.ok n =>
            have p := s.nextPath;
            {
              atts :=
                setAtt s.atts id fun x =>
                  { id := x.id, upload := x.upload, size := x.size, plainLen := x.plainLen, pad := x.pad,
                    bucket := x.bucket, hasDigest := x.hasDigest, hasHash := x.hasHash, st := St.published,
                    path := some p },
              files := s.files ++ [{ path := p, content := r.genuine, intact := true, len := n }], nextPath := p + 1,
              maxSize := s.maxSize, delivered := s.delivered, fetched := s.fetched }
          | Except.error a =>
            {
              atts :=
                setAtt s.atts id fun x =>
                  { id := x.id, upload := x.upload, size := x.size, plainLen := x.plainLen, pad := x.pad,
                    bucket := x.bucket, hasDigest := x.hasDigest, hasHash := x.hasHash, st := St.error,
                    path := x.path },
              files := s.files, nextPath := s.nextPath, maxSize := s.maxSize, delivered := s.delivered,
              fetched := s.fetched }
def Attachments.consume : S → Nat → S :=
fun s id =>
  match List.find? (fun x => x.id == id) s.atts with
  |
  some
      { id := id_1, upload := upload, size := size, plainLen := plainLen, pad := pad, bucket := bucket,
        hasDigest := hasDigest, hasHash := hasHash, st := St.published, path := some p } =>
    match List.find? (fun x => x.path == p) s.files with
    | some f =>
      if f.intact = true then
        { atts := s.atts, files := s.files, nextPath := s.nextPath, maxSize := s.maxSize,
          delivered := s.delivered ++ [(id, f.content, f.len)], fetched := s.fetched }
      else s
    | none => s
  | x => s
def Attachments.step : S → Ev → S :=
fun s x =>
  match x with
  | Ev.arrive a =>
    if (s.atts.any fun x => x.id == a.id) = true then s
    else
      {
        atts :=
          s.atts ++
            [{ id := a.id, upload := a.upload, size := a.size, plainLen := a.plainLen, pad := a.pad, bucket := a.bucket,
                hasDigest := a.hasDigest, hasHash := a.hasHash, st := St.pending, path := none }],
        files := s.files, nextPath := s.nextPath, maxSize := s.maxSize, delivered := s.delivered, fetched := s.fetched }
  | Ev.download id r => download s id r
  | Ev.consume id => consume s id
  | Ev.tamper p =>
    { atts := s.atts,
      files :=
        List.map
          (fun f =>
            if (f.path == p) = true then { path := f.path, content := f.content, intact := false, len := f.len } else f)
          s.files,
      nextPath := s.nextPath, maxSize := s.maxSize, delivered := s.delivered, fetched := s.fetched }

def Attachments.Delivered : S → Prop :=
fun s =>
  ∀ (x : Nat × Option Nat × Nat),
    x ∈ s.delivered → ∃ a, a ∈ s.atts ∧ a.id = x.fst ∧ x.snd.fst = some a.upload ∧ x.snd.snd ≤ a.size
def Attachments.Published : S → Prop :=
fun s =>
  ∀ (a : Att),
    a ∈ s.atts →
      (a.path.isSome = true ↔ a.st = St.published) ∧
        ∀ (p : Nat), a.path = some p → ∃ f, f ∈ s.files ∧ f.path = p ∧ f.content = some a.upload ∧ f.len ≤ a.size
def Attachments.Fetched : S → Prop :=
fun s =>
  ∀ (x : Nat × Nat),
    x ∈ s.fetched → ∃ a, a ∈ s.atts ∧ a.id = x.fst ∧ x.snd ≤ ciphertextSize a.bucket + margin + 1 ∧ a.size ≤ s.maxSize

decrypt_ok : ∀ (b : Blob) (n : Nat),
  decrypt b = Except.ok n →
    b.keyOK = true ∧
      shapeOK b.len = true ∧
        b.macOK = true ∧
          (b.inc = Inc.none ∨ b.inc = Inc.ok) ∧
            (b.integrity = Integrity.macOnly ∨
                b.integrity = Integrity.hash true ∨ b.integrity = Integrity.digest false true) ∧
              n ≤ b.size ∧ ∃ p, b.plain = some p ∧ n ≤ p
describe_ok : ∀ (a : Att) (r : Resp) (n : Nat),
  decrypt (describe a r) = Except.ok n →
    r.genuine = some a.upload ∧ (a.hasDigest = true ∨ a.hasHash = true) ∧ n ≤ a.size
