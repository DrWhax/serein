def CallCapture.leavesActive : S → S → Bool :=
fun s s' =>
  s'.active == s.active && s'.gen == s.gen && s'.share == s.share && s'.localVideo == s.localVideo &&
      s'.setupTask == s.setupTask &&
    !(effects s').any fun x =>
        match x with
        | Eff.svcHangup excludeRinging => true
        | Eff.svcEndCall c id =>
          match s.ncall, s.active with
          | some n, some a => n.conv == c && Option.all (fun x => x == n.id) id && s.nativeInst n.id == some a.inst
          | x, x_1 => false
        | x => false

def CallCapture.evShare : S → Ev → Option Nat :=
fun s x =>
  match x with
  | Ev.srcDone i ok => some i
  | Ev.scFrame j => streamOwner s j
  | Ev.scStopped j => streamOwner s j
  | Ev.scStartDone j ok => streamOwner s j
  | Ev.scStopDone j ok => streamOwner s j
  | Ev.run k =>
    match s.jobs[k]? with
    | some (Job.capStop i) => some i
    | some (Job.sources i tok gen c) => some i
    | some (Job.present i pt) => some i
    | some (Job.sourcesRet i tok gen c ok) => some i
    | some (Job.installRet i cg j pt ok) => some i
    | some (Job.teardownRet i cg j pt) => some i
    | some (Job.scFrame i tok) => some i
    | some (Job.scStopped i tok) => some i
    | some (Job.stopRet j) => streamOwner s j
    | x => none
  | x => none
def CallCapture.t2Stop : S → Ev → S → Bool :=
fun s e s' =>
  match evShare s e, s.share with
  | some i, some (m, snd) =>
    if (i == m) = true then true
    else
      s'.share == s.share && s'.presenting == s.presenting &&
        (effects s').all fun x =>
          match x with
          | Eff.scStopReq j => streamOwner s j != some m
          | Eff.svcSetPresenting c on => false
          | x => true
  | x, x_1 => true

def CallCapture.uiEnabled : S → Ev → Bool :=
fun s x =>
  match x with
  | Ev.openLobby c video => s.active.isNone
  | Ev.startCall =>
    match s.active with
    | some a => a.stage == Stage.lobby && a.live && !s.setupTask
    | none => false
  | Ev.accept c video => Option.map (fun x => x.fst) s.ringing == some c && s.active.isNone
  | Ev.decline c => Option.map (fun x => x.fst) s.ringing == some c && s.active.isNone
  | Ev.hangUp =>
    match s.active with
    | some a => a.stage == Stage.inCall && a.live
    | none => false
  | Ev.closeCall =>
    match s.active with
    | some a => a.stage == Stage.lobby || !a.live
    | none => false
  | Ev.setVideo on =>
    match s.active with
    | some a => a.live
    | none => false
  | Ev.toggleShare =>
    match s.active with
    | some a => a.stage == Stage.inCall && a.live
    | none => false
  | Ev.pickSource => s.sources
  | Ev.cancelPick => s.sources
  | x => true
