def Modifiers.accepts : World → Row → Modifiers.Mod → Bool :=
fun w r m =>
  match m.kind with
  | Kind.delete =>
    !r.deleted && inConv r m && r.conv != w.us && m.sender == r.author && within m.server r.server deleteWindow
  | Kind.adminDelete =>
    !r.deleted && inConv r m && r.conv != w.us && isAdmin w r.conv m.sender && within m.server r.server deleteWindow
  | Kind.edit => editValid w r m && decide (r.rev < m.ts)
  | Kind.reaction =>
    !r.deleted && !r.viewOnce &&
      (m.sender == w.us ||
        r.conv == m.conv && belongs w m.sender r.conv && !blockedIn w r.conv m.sender && sentToSender r m)
  | Kind.vote =>
    !r.deleted && r.poll && !r.terminated && belongs w m.sender r.conv &&
      (m.sender == w.us || inConv r m && !blockedIn w r.conv m.sender && sentToSender r m)
  | Kind.terminate => r.poll && !r.terminated && m.sender == r.author && inConv r m
  | Kind.pin => pinnable w r m
  | Kind.unpin => pinnable w r m && r.pinned
def Modifiers.editValid : World → Row → Modifiers.Mod → Bool :=
fun w r m =>
  editable r && m.sender == r.author && r.conv == m.conv &&
      (within m.server r.server editWindow || m.src != Src.other) &&
    (m.src != Src.other || !blockedIn w r.conv m.sender)
def Modifiers.pinnable : World → Row → Modifiers.Mod → Bool :=
fun w r m =>
  !r.deleted && !r.giftBadge && inConv r m && !blockedIn w r.conv m.sender && belongs w m.sender r.conv &&
    canEditAttrs w r.conv m.sender
def Modifiers.belongs : World → Nat → Nat → Bool :=
fun w s c =>
  s == w.us ||
    match w.group c with
    | some g => g.members.contains s
    | none => c == s
def Modifiers.blockedIn : World → Nat → Nat → Bool :=
fun w c s =>
  w.blocked.contains s ||
    match w.group c with
    | some val => w.blockedGroups.contains c
    | none => w.blocked.contains c
def Modifiers.canEditAttrs : World → Nat → Nat → Bool :=
fun w c s =>
  match w.group c with
  | some g => g.members.contains s && (g.admins.contains s || g.openAttrs)
  | none => true

def Modifiers.resolves : Row → Modifiers.Mod → Bool :=
fun r m => r.author == m.author && (r.sentAt == m.target || r.revs.contains m.target)
def Modifiers.lookup : List Row → Modifiers.Mod → Option Row :=
fun rows m =>

    List.find? (fun r => r.author == m.author && r.revs.contains m.target) rows

def Modifiers.kept : List Entry → Entry → Nat → Nat → List Entry :=

  have l' := List.filter (fun x => decide (now - x.stored ≤ pendingMaxAge)) l ++ [e];
  trim limit l'.length l'

def Modifiers.step : World → S → Ev → S :=
fun w s x =>
  match x with
  | Ev.recv r =>
    if (s.rows.any fun x => x.author == r.author && x.sentAt == r.sentAt) = true then s
    else
      have r := fresh s r;
      have k := (r.author, r.sentAt);
      have s1 :=
        { rows := s.rows ++ [r], pending := s.pending, next := s.next + 1, now := s.now, resolved := s.resolved,
          exhausted := s.exhausted };
      List.foldl (fun s e => applyMod w (s1.pending.length + 1) s e.mod false)
        { rows := s1.rows, pending := List.filter (fun x => !targets k x) s1.pending, next := s1.next, now := s1.now,
          resolved := s1.resolved, exhausted := s1.exhausted }
        (sortTs (List.filter (targets k) s1.pending))
  | Ev.mod m => applyMod w (s.pending.length + 2) s m (m.src != Src.here)
  | Ev.forget a t =>
    { rows := List.filter (fun x => !(x.author == a && x.sentAt == t)) s.rows, pending := s.pending, next := s.next,
      now := s.now, resolved := s.resolved, exhausted := s.exhausted }
  | Ev.tick n =>
    { rows := s.rows, pending := s.pending, next := s.next, now := s.now + n, resolved := s.resolved,
      exhausted := s.exhausted }

def Modifiers.ResolvedCorrect : World → S → Prop :=
fun w s =>
  ∀ (x : Modifiers.Mod × Row × Bool),
    x ∈ s.resolved → resolves x.snd.fst x.fst = true ∧ x.snd.snd = accepts w x.snd.fst x.fst
def Modifiers.PendingUnresolvable : S → Prop :=
fun s => ∀ (e : Entry), e ∈ s.pending → ∀ (r : Row), r ∈ s.rows → resolves r e.mod = false
def Modifiers.NewestShown : World → S → Prop :=
fun w s =>
  ∀ (r : Row),
    r ∈ s.rows →
      (r.deleted = false →
          ∀ (x : Modifiers.Mod × Row × Bool),
            x ∈ s.resolved →
              x.snd.fst.gen = r.gen → x.fst.kind = Kind.edit → editValid w x.snd.fst x.fst = true → x.fst.ts ≤ r.rev) ∧
        (r.rev = r.sentAt ∨
          ∃ x, x ∈ s.resolved ∧ x.snd.fst.gen = r.gen ∧ x.fst.kind = Kind.edit ∧ x.snd.snd = true ∧ x.fst.ts = r.rev)
def Modifiers.StepOk : S → S → Prop :=
fun s s' =>
  ∃ new,
    s'.resolved = new ++ s.resolved ∧
      ∀ (g : Nat) (r : Row),
        s.get g = some r →
          s'.get g = none ∨
            ∃ r',
              s'.get g = some r' ∧
                r'.author = r.author ∧
                  r'.sentAt = r.sentAt ∧
                    r'.conv = r.conv ∧
                      r.rev ≤ r'.rev ∧
                        (r.deleted = true → r' = r) ∧ (r' ≠ r → ∃ x, x ∈ new ∧ x.snd.fst.gen = g ∧ x.snd.snd = true)
