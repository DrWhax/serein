def Send.Content.carries : Content → Bool :=
fun x =>
  match x with
  | Content.data d => d.extra
  | Content.sent m update story edit => (Option.map (fun x => x.extra) m).getD false || story || edit
  | Content.content => true
  | Content.other => false
def Send.Data.onlyDeletes : Data → Bool :=
fun d => d.retract && !d.extra
def Send.Content.onlyRetracts : Content → Bool :=
fun x =>
  match x with
  | Content.data d => d.onlyDeletes
  | Content.sent (some d) update story edit => d.onlyDeletes && !story && !edit
  | Content.sent none update story edit => update && !story && !edit
  | x => false

def Send.S.blocked : S → Nat → Bool :=
fun s p => s.blockedIds.contains p || s.blockedNums.contains p
def Send.authorized : S → Target → Content → List Nat → Bool :=
fun s t c ids =>
  (match t with
    | Target.direct p => !s.blocked p
    | Target.self => true
    | Target.group g => !s.blockedGroups.contains g) &&
    (c.onlyRetracts || ids.all fun m => s.msgs m == Msg.live)
def Send.recipients : S → Nat → List Nat :=
fun s g => List.filter (fun p => !s.blocked p) (s.members g)
def Send.stepCore : P → Ev → P :=
fun p x =>
  match x with
  | Ev.queue t c ids =>
    { s := p.s, jobs := p.jobs ++ [{ target := t, content := c, ids := ids, phase := Phase.queued }], out := p.out,
      refused := p.refused }
  | Ev.start j =>
    match p.jobs[j]? with
    | some x =>
      if (x.phase != Phase.queued) = true then p
      else
        if authorized p.s x.target x.content x.ids = true then
          have to :=
            match x.target with
            | Target.group g => recipients p.s g
            | x => [];
          { s := p.s,
            jobs := p.jobs.set j { target := x.target, content := x.content, ids := x.ids, phase := Phase.started to },
            out := p.out, refused := p.refused }
        else { s := p.s, jobs := p.jobs.eraseIdx j, out := p.out, refused := p.refused + 1 }
    | none => p
  | Ev.submit j =>
    match p.jobs[j]? with
    | some { target := t, content := c, ids := ids, phase := Phase.started to } =>
      if authorized p.s t c ids = true then
        { s := p.s, jobs := p.jobs.eraseIdx j, out := p.out ++ [{ target := t, content := c, ids := ids, to := to }],
          refused := p.refused }
      else { s := p.s, jobs := p.jobs.eraseIdx j, out := p.out, refused := p.refused + 1 }
    | x => p
  | Ev.block q b =>
    {
      s :=
        have __src := p.s;
        { blockedIds := setIn p.s.blockedIds q b, blockedNums := __src.blockedNums,
          blockedGroups := __src.blockedGroups, msgs := __src.msgs, members := __src.members },
      jobs := p.jobs, out := p.out, refused := p.refused }
  | Ev.blockNumber q b =>
    {
      s :=
        have __src := p.s;
        { blockedIds := __src.blockedIds, blockedNums := setIn p.s.blockedNums q b,
          blockedGroups := __src.blockedGroups, msgs := __src.msgs, members := __src.members },
      jobs := p.jobs, out := p.out, refused := p.refused }
  | Ev.blockGroup g b =>
    {
      s :=
        have __src := p.s;
        { blockedIds := __src.blockedIds, blockedNums := __src.blockedNums,
          blockedGroups := setIn p.s.blockedGroups g b, msgs := __src.msgs, members := __src.members },
      jobs := p.jobs, out := p.out, refused := p.refused }
  | Ev.erase m =>
    {
      s :=
        have __src := p.s;
        { blockedIds := __src.blockedIds, blockedNums := __src.blockedNums, blockedGroups := __src.blockedGroups,
          msgs := upd p.s.msgs m Msg.erased, members := __src.members },
      jobs := p.jobs, out := p.out, refused := p.refused }
  | Ev.remove m =>
    {
      s :=
        have __src := p.s;
        { blockedIds := __src.blockedIds, blockedNums := __src.blockedNums, blockedGroups := __src.blockedGroups,
          msgs := upd p.s.msgs m Msg.gone, members := __src.members },
      jobs := p.jobs, out := p.out, refused := p.refused }
def Send.submitted : P → P → Option Out :=
fun p p' => if p'.out.length = p.out.length + 1 then p'.out.getLast? else none
def Send.submitAuthorized : P → Ev → P → Bool :=
fun p _e p' =>
  (p'.out.length == p.out.length || p'.out.length == p.out.length + 1) &&
    match submitted p p' with
    | some o => authorized p.s o.target o.content o.ids
    | none => true
def Send.contentNeedsSource : P → Ev → P → Bool :=
fun p _e p' =>
  match submitted p p' with
  | some o => !o.content.carries || o.ids.all fun m => p.s.msgs m == Msg.live
  | none => true
def Send.retractionGoes : P → Ev → P → Bool :=
fun p e p' =>
  match e with
  | Ev.submit j =>
    match p.jobs[j]? with
    | some { target := t, content := c, ids := ids, phase := Phase.started to } =>
      !c.onlyRetracts ||
          !authorized
              (have __src := p.s;
              { blockedIds := __src.blockedIds, blockedNums := __src.blockedNums, blockedGroups := __src.blockedGroups,
                msgs := fun x => Msg.live, members := __src.members })
              t c ids ||
        p'.out.length == p.out.length + 1
    | x => true
  | x => true
def Send.groupRecipients : P → Ev → P → Bool :=
fun p e p' =>
  match e with
  | Ev.start j =>
    match p.jobs[j]? with
    | some { target := Target.group g, content := c, ids := ids, phase := Phase.queued } =>
      !authorized p.s (Target.group g) c ids ||
        p'.jobs[j]? ==
          some { target := Target.group g, content := c, ids := ids, phase := Phase.started (recipients p.s g) }
    | x => true
  | x => true
