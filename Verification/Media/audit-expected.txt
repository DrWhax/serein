def Media.live : Db → Nat → Bool :=
fun d m =>
  match d.rows m with
  | some r => !r.erased && Option.all (fun x => decide (x > d.now)) r.expires
  | none => false
def Media.permits : Db → Nat → Nat → Bool :=
fun d m p => live d m && d.rel.contains (m, p)
def Media.permitsAny : Db → Nat → Bool :=
fun d p => d.rel.any fun x => x.snd == p && live d x.fst
def Media.avail : Db → List (Nat × Nat) :=
fun d => List.filter (fun x => live d x.fst) d.rel

number of parameters: 0

def Media.apply : List (Nat × Nat) → UI → UI :=
fun a u =>
  {
    lightbox :=
      Option.filter
        (fun x =>
          match x with
          | (m, ps) => (a.any fun x => x.fst == m) && ps.all fun p => a.contains (m, p))
        u.lightbox,
    video := Option.filter (fun x => a.contains x) u.video, audio := Option.filter (fun x => a.contains x) u.audio,
    cache := List.filter (fun p => a.any fun x => x.snd == p) u.cache }
def Media.stepCore : S → Ev → S :=
fun s x =>
  match x with
  | Ev.openLightbox m ps =>
    if ps.all (permits s.d m) = true then
      { d := s.d,
        u :=
          have __src := s.u;
          { lightbox := some (m, ps), video := __src.video, audio := __src.audio, cache := __src.cache },
        decodes := s.decodes, save := s.save, out := s.out }
    else s
  | Ev.playVideo m p =>
    if permits s.d m p = true then
      { d := s.d,
        u :=
          have __src := s.u;
          { lightbox := __src.lightbox, video := some (m, p), audio := __src.audio, cache := __src.cache },
        decodes := s.decodes, save := s.save, out := s.out ++ [("video", m, p)] }
    else s
  | Ev.playAudio m p =>
    if permits s.d m p = true then
      { d := s.d,
        u :=
          have __src := s.u;
          { lightbox := __src.lightbox, video := __src.video, audio := some (m, p), cache := __src.cache },
        decodes := s.decodes, save := s.save, out := s.out ++ [("audio", m, p)] }
    else s
  | Ev.decode m p => { d := s.d, u := s.u, decodes := s.decodes ++ [(m, p)], save := s.save, out := s.out }
  | Ev.decoded k =>
    match s.decodes[k]? with
    | some (m, p) =>
      have s := { d := s.d, u := s.u, decodes := s.decodes.eraseIdx k, save := s.save, out := s.out };
      if permits s.d m p = true then
        { d := s.d,
          u :=
            have __src := s.u;
            { lightbox := __src.lightbox, video := __src.video, audio := __src.audio, cache := p :: s.u.cache },
          decodes := s.decodes, save := s.save, out := s.out }
      else s
    | none => s
  | Ev.openSave m p =>
    if permits s.d m p = true then { d := s.d, u := s.u, decodes := s.decodes, save := some (m, p), out := s.out }
    else s
  | Ev.confirmSave =>
    match s.save with
    | some (m, p) =>
      if permits s.d m p = true then
        { d := s.d, u := s.u, decodes := s.decodes, save := none, out := s.out ++ [("export", m, p)] }
      else { d := s.d, u := s.u, decodes := s.decodes, save := none, out := s.out }
    | none => s
  | Ev.delete m =>
    {
      d :=
        have __src := s.d;
        { rows := upd s.d.rows m none, rel := __src.rel, now := __src.now },
      u := s.u, decodes := s.decodes, save := s.save, out := s.out }
  | Ev.erase m =>
    {
      d :=
        have __src := s.d;
        { rows := upd s.d.rows m (Option.map (fun r => { erased := true, expires := r.expires }) (s.d.rows m)),
          rel := __src.rel, now := __src.now },
      u := s.u, decodes := s.decodes, save := s.save, out := s.out }
  | Ev.expireAt m t =>
    {
      d :=
        have __src := s.d;
        { rows := upd s.d.rows m (Option.map (fun r => { erased := r.erased, expires := some t }) (s.d.rows m)),
          rel := __src.rel, now := __src.now },
      u := s.u, decodes := s.decodes, save := s.save, out := s.out }
  | Ev.tick =>
    {
      d :=
        have __src := s.d;
        { rows := __src.rows, rel := __src.rel, now := s.d.now + 1 },
      u := s.u, decodes := s.decodes, save := s.save, out := s.out }
  | Ev.ref m p =>
    {
      d :=
        have __src := s.d;
        {
          rows :=
            if (s.d.rows m).isSome = true then s.d.rows else upd s.d.rows m (some { erased := false, expires := none }),
          rel := (m, p) :: s.d.rel, now := __src.now },
      u := s.u, decodes := s.decodes, save := s.save, out := s.out }
  | Ev.update => { d := s.d, u := apply (avail s.d) s.u, decodes := s.decodes, save := s.save, out := s.out }
def Media.handedOutPermitted : S → Ev → S → Bool :=
fun s _e s' =>
  decide (s'.out.length ≥ s.out.length) &&
    (newOut s s').all fun x =>
      match x with
      | (fst, m, p) => permits s.d m p
def Media.decodePermitted : S → Ev → S → Bool :=
fun s e s' =>
  match e with
  | Ev.decoded k =>
    match s.decodes[k]? with
    | some (m, p) => s'.u.cache == s.u.cache || s'.u.cache == p :: s.u.cache && permits s.d m p
    | none => s'.u.cache == s.u.cache
  | Ev.update => s'.u.cache.all fun x => s.u.cache.contains x
  | x => s'.u.cache == s.u.cache
def Media.updateRevokes : S → Ev → S → Bool :=
fun _s e s' =>
  match e with
  | Ev.update =>
    Option.all
            (fun x =>
              match x with
              | (m, ps) => live s'.d m && ps.all fun p => permits s'.d m p)
            s'.u.lightbox &&
          Option.all
            (fun x =>
              match x with
              | (m, p) => permits s'.d m p)
            s'.u.video &&
        Option.all
          (fun x =>
            match x with
            | (m, p) => permits s'.d m p)
          s'.u.audio &&
      s'.u.cache.all (permitsAny s'.d)
  | x => true
