def Codecs.need : Cursor → Int → Bool :=
fun c n => decide (0 ≤ n) && decide (n ≤ ↑(c.stop - c.pos))
def Codecs.be : List Nat → Nat → Nat → Nat :=
fun bs p k => List.foldl (fun acc i => acc * 256 + byteAt bs (p + i)) 0 (List.range k)
def Codecs.le : List Nat → Nat → Nat → Nat :=
fun bs p k => List.foldr (fun i acc => acc * 256 + byteAt bs (p + i)) 0 (List.range k)
def Codecs.fixedRead : List Nat → Cursor → Nat → Bool → Except Fail (Val × Cursor) :=
fun bs c k big =>
  if need c ↑k = true then Except.ok (Val.num (if big = true then be bs c.pos k else le bs c.pos k), advance c k)
  else Except.error { offset := c.pos, kind := Kind.unexpectedEnd }
def Codecs.prim : List Nat → Cursor → Prim → Except Fail (Val × Cursor) :=
fun bs c x =>
  match x with
  | Prim.u8 => fixedRead bs c 1 true
  | Prim.u16be => fixedRead bs c 2 true
  | Prim.u16le => fixedRead bs c 2 false
  | Prim.u24be => fixedRead bs c 3 true
  | Prim.u24le => fixedRead bs c 3 false
  | Prim.u32be => fixedRead bs c 4 true
  | Prim.u32le => fixedRead bs c 4 false
  | Prim.u64be =>
    match fixedRead bs c 4 true with
    | Except.ok (Val.num hi, c1) =>
      match fixedRead bs c1 4 true with
      | Except.ok (Val.num lo, c2) => Except.ok (Val.num (hi * 4294967296 + lo), c2)
      | Except.ok a => Except.error { offset := c1.pos, kind := Kind.invalid }
      | Except.error e => Except.error e
    | Except.ok a => Except.error { offset := c.pos, kind := Kind.invalid }
    | Except.error e => Except.error e
  | Prim.bytes n =>
    if need c n = true then Except.ok (Val.slice c.pos (c.pos + n.toNat), advance c n.toNat)
    else Except.error { offset := c.pos, kind := Kind.unexpectedEnd }
  | Prim.skip n =>
    if need c n = true then Except.ok (Val.unit, advance c n.toNat)
    else Except.error { offset := c.pos, kind := Kind.unexpectedEnd }
  | Prim.expect lit =>
    if (!need c ↑lit.length) = true then Except.error { offset := c.pos, kind := Kind.unexpectedEnd }
    else
      if ((List.range lit.length).all fun i => byteAt bs (c.pos + i) == lit.getD i 0) = true then
        Except.ok (Val.unit, advance c lit.length)
      else Except.error { offset := c.pos, kind := Kind.mismatch }
  | Prim.rest => Except.ok (Val.slice c.pos c.stop, { start := c.start, pos := c.stop, stop := c.stop })

def Codecs.header : List Nat → Nat → Nat → Bool → Except Fail (Nat × Nat × Nat) :=
fun bs pos hi top =>
  have size32 := be bs pos 4;
  have type := be bs (pos + 4) 4;
  have p := pos + 8;
  have sized :=
    if (size32 == 1) = true then
      if hi - p < 4 then Except.error { offset := p, kind := Kind.unexpectedEnd }
      else
        if hi - p < 8 then Except.error { offset := p + 4, kind := Kind.unexpectedEnd }
        else Except.ok (p + 8, be bs p 8)
    else
      if (size32 == 0) = true then
        if top = true then Except.ok (p, hi - p + 8) else Except.error { offset := p, kind := Kind.invalid }
      else Except.ok (p, size32);
  match sized with
  | Except.error e => Except.error e
  | Except.ok (p, size) =>
    if (type == uuid) = true then
      if hi - p < 16 then Except.error { offset := p, kind := Kind.unexpectedEnd } else Except.ok (type, p + 16, size)
    else Except.ok (type, p, size)

def Codecs.bitAt : List Nat → Nat → Nat :=

def Codecs.TIFF.entry : List Nat → Bool → TIFF.Fields → Cursor → Except Fail (TIFF.Fields × Cursor) :=
fun bs big fs c => do
  let __x ← TIFF.rd bs big 2 c
  match __x with
    | (code, c) => do
      let __x ← TIFF.rd bs big 2 c
      match __x with
        | (type, c) => do
          let __x ← TIFF.rd bs big 4 c
          match __x with
            | (count, c) =>
              have slot := c.pos;
              do
              TIFF.check (need c 4) { offset := c.pos, kind := Kind.unexpectedEnd }
              have c : Cursor := advance c 4
              match TIFF.Tag.ofCode code with
                | none => pure (fs, c)
                | some t => do
                  TIFF.check ((fs t).isNone && (type == 3 || type == 4) && decide (0 < count))
                      { offset := c.pos, kind := Kind.invalid }
                  have size : Nat := count * if (type == 3) = true then 2 else 4
                  if size ≤ 4 then pure (fs.set t { short := type == 3, count := count, lo := slot }, c)
                    else
                      have voff := if big = true then be bs slot 4 else le bs slot 4;
                      do
                      TIFF.check (decide (voff ≤ bs.length)) { offset := 0, kind := Kind.unexpectedEnd }
                      TIFF.check (decide (size ≤ bs.length - voff)) { offset := voff, kind := Kind.unexpectedEnd }
                      pure (fs.set t { short := type == 3, count := count, lo := voff }, c)

def Codecs.TIFF.directory : List Nat → Except Fail TIFF.Dir :=
fun bs => do
  let __x ← TIFF.rd bs true 2 { start := 0, pos := 0, stop := bs.length }
  match __x with
    | (mark, c) => do
      let big ← TIFF.order mark c.pos
      let __x ← TIFF.rd bs big 2 c
      match __x with
        | (magic, c) => do
          TIFF.check (magic == 42) { offset := c.pos, kind := Kind.invalid }
          let __x ← TIFF.rd bs big 4 c
          match __x with
            | (off, snd) => do
              TIFF.check (need { start := 0, pos := 0, stop := bs.length } ↑off)
                  { offset := 0, kind := Kind.unexpectedEnd }
              let __x ← TIFF.rd bs big 2 { start := 0, pos := off, stop := bs.length }
              match __x with
                | (count, e) => do
                  let __x ← TIFF.entries bs big count (fun x => none) e
                  match __x with
                    | (fs, e) => do
                      TIFF.check (need e 4) { offset := e.pos, kind := Kind.unexpectedEnd }
                      pure { big := big, fields := fs }
def Codecs.TIFF.values : List Nat → TIFF.Dir → TIFF.Tag → Nat → Except Fail (List Nat) :=
fun bs d t count =>
  match d.fields t with
  | some f =>
    if (f.count == count) = true then
      Except.ok
        (List.map
          (fun i => if d.big = true then be bs (f.lo + i * f.width) f.width else le bs (f.lo + i * f.width) f.width)
          (List.range count))
    else Except.error TIFF.bad
  | none => Except.error TIFF.bad

def Codecs.TIFF.checkDims : TIFF.Limits → Nat → Nat → Except Fail Unit :=
fun lim w h =>
  if (w == 0 || h == 0) = true then Except.error TIFF.bad
  else
    if (decide (lim.maxDimension < w) || decide (lim.maxDimension < h) || decide (lim.maxPixels < w * h)) = true then
      Except.error { offset := 0, kind := Kind.limit }
    else Except.ok ()
def Codecs.TIFF.defaults : List Nat → TIFF.Dir → Except Fail Unit :=
fun bs d => do
  forIn [TIFF.Tag.compression, TIFF.Tag.predictor, TIFF.Tag.orientation, TIFF.Tag.fillOrder, TIFF.Tag.planar] PUnit.unit
      fun t __s => do
      let v ← TIFF.scalar bs d t (some 1)
      TIFF.check (v == 1) TIFF.bad
      pure (ForInStep.yield PUnit.unit)
  pure ()
def Codecs.TIFF.unsigned : List Nat → TIFF.Dir → Nat → Except Fail Unit :=
fun bs d samples =>
  if (d.fields TIFF.Tag.sampleFormat).isSome = true then do
    let formats ← TIFF.values bs d TIFF.Tag.sampleFormat samples
    TIFF.check (formats.all fun x => x == 1) TIFF.bad
  else Except.ok ()

def Codecs.TIFF.layout : List Nat → TIFF.Limits → Except Fail TIFF.Layout :=
fun bs lim => do
  TIFF.check (decide (bs.length ≤ lim.maxInput)) { offset := 0, kind := Kind.limit }
  let d ← TIFF.directory bs
  let width ← TIFF.scalar bs d TIFF.Tag.width none
  let height ← TIFF.scalar bs d TIFF.Tag.height none
  TIFF.checkDims lim width height
  let photo ← TIFF.scalar bs d TIFF.Tag.photometric none
  let samples ← TIFF.scalar bs d TIFF.Tag.samples (some 1)
  have colors : Nat := if (photo == 2) = true then 3 else 1
  TIFF.check (decide (photo ≤ 2) && (samples == colors || samples == colors + 1)) TIFF.bad
  TIFF.defaults bs d
  let bits ← TIFF.values bs d TIFF.Tag.bits samples
  have b0 : Nat := bits.headD 0
  TIFF.check ((b0 == 8 || b0 == 16) && bits.all fun x => x == b0) TIFF.bad
  TIFF.unsigned bs d samples
  let extra ← TIFF.scalar bs d TIFF.Tag.extraSamples (some 0)
  TIFF.check (if colors < samples then decide (extra ≤ 2) else extra == 0) TIFF.bad
  let rows ← TIFF.scalar bs d TIFF.Tag.rows (some height)
  TIFF.check (decide (0 < rows)) TIFF.bad
  have count : Nat := 1 + (height - 1) / rows
  let offsets ← TIFF.values bs d TIFF.Tag.offsets count
  let sizes ← TIFF.values bs d TIFF.Tag.byteCounts count
  have stride : Nat := width * samples * (b0 / 8)
  let ss ← TIFF.strips bs.length height rows stride 0 (offsets.zip sizes)
  pure
      { big := d.big, width := width, height := height, photometric := photo, samples := samples, bits := b0,
        extra := extra, strips := ss }
def Codecs.TIFF.pixel : List Nat → TIFF.Layout → Nat → List Nat :=
fun bs L p =>
  have w := L.bits / 8;
  have first := TIFF.sample bs L p;
  have red := if (L.photometric == 0) = true then 255 - first else first;
  have green := if (L.colors == 3) = true then TIFF.sample bs L (p + w) else red;
  have blue := if (L.colors == 3) = true then TIFF.sample bs L (p + 2 * w) else red;
  have last := if L.colors < L.samples then TIFF.sample bs L (p + L.colors * w) else 255;
  have alpha := if (L.extra == 0) = true then 255 else last;
  [TIFF.straight L red alpha, TIFF.straight L green alpha, TIFF.straight L blue alpha, alpha]
def Codecs.TIFF.stripPixels : List Nat → TIFF.Layout → Nat → Nat → List Nat :=

def Codecs.TIFF.pixels : List Nat → TIFF.Layout → List Nat :=

@need_spec : ∀ {c : Cursor} {n : Int}, Good c → need c n = true → 0 ≤ n ∧ c.pos + n.toNat ≤ c.stop
@prim_within : ∀ {bs : List Nat} {c : Cursor} {p : Prim} {v : Val} {c' : Cursor},
  Good c →
    prim bs c p = Except.ok (v, c') →
      c'.start = c.start ∧
        c'.stop = c.stop ∧
          c.pos ≤ c'.pos ∧ c'.pos ≤ c.stop ∧ ∀ (lo hi : Nat), v = Val.slice lo hi → c.pos ≤ lo ∧ lo ≤ hi ∧ hi ≤ c'.pos
run_within : ∀ (bs : List Nat) (g : G) (c : Cursor), Good c → ∀ (r : R), run bs g c = Except.ok r → Within c r
region_exact : ∀ (bs : List Nat) (n : Int) (g : G) (c : Cursor),
  Good c → ∀ (r : R), run bs (G.region n g) c = Except.ok r → 0 ≤ n ∧ r.c.pos = c.pos + n.toNat
run_budget : ∀ (bs : List Nat) (g : G) (c : Cursor) (r : R), run bs g c = Except.ok r → r.items ≤ budget g
children_within : ∀ (bs : List Nat) (lo hi : Nat) (top : Bool) (count maxBoxes : Nat),
  lo ≤ hi →
    ∀ (bxs : List Box) (n : Nat), children bs lo hi top count maxBoxes = Except.ok (bxs, n) → BoxesWithin lo hi bxs
children_budget : ∀ (bs : List Nat) (lo hi : Nat) (top : Bool) (count maxBoxes : Nat),
  lo ≤ hi →
    ∀ (bxs : List Box) (n : Nat),
      children bs lo hi top count maxBoxes = Except.ok (bxs, n) → n = count + bxs.length ∧ (bxs ≠ [] → n ≤ maxBoxes)
bitsVal_lt : ∀ (bs : List Nat) (p n : Nat), bitsVal bs p n < 2 ^ n
readBits_spec : ∀ (bs : List Nat) (zeroFill : Bool) (r : Bits) (n v : Nat) (r' : Bits),
  r.pos ≤ 8 * bs.length →
    readBits bs zeroFill r n = Except.ok (v, r') →
      v < 2 ^ n ∧
        r'.pos ≤ 8 * bs.length ∧
          r.pos ≤ r'.pos ∧ (r'.pos = r.pos + n ∨ zeroFill = true ∧ r'.exhausted = true ∧ r'.pos = 8 * bs.length)
