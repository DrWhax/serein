def Codecs.need : Cursor → Int → Bool :=
fun c n => decide (0 ≤ n) && decide (n ≤ ↑(c.stop - c.pos))
def Codecs.be : List Nat → Nat → Nat → Nat :=
fun bs p k => List.foldl (fun acc i => acc * 256 + byteAt bs (p + i)) 0 (List.range k)
def Codecs.le : List Nat → Nat → Nat → Nat :=
fun bs p k => List.foldr (fun i acc => acc * 256 + byteAt bs (p + i)) 0 (List.range k)
def Codecs.fixedRead : List Nat → Cursor → Nat → Bool → Except Fail (Val × Cursor) :=
fun bs c k big =>
  if need c ↑k = true then Except.ok (Val.num (if big = true then be bs c.pos k else le bs c.pos k), advance c k)
  else Except.error { offset := c.pos, kind := Kind.unexpectedEnd }
def Codecs.prim : List Nat → Cursor → Prim → Except Fail (Val × Cursor) :=
fun bs c x =>
  match x with
  | Prim.u8 => fixedRead bs c 1 true
  | Prim.u16be => fixedRead bs c 2 true
  | Prim.u16le => fixedRead bs c 2 false
  | Prim.u24be => fixedRead bs c 3 true
  | Prim.u24le => fixedRead bs c 3 false
  | Prim.u32be => fixedRead bs c 4 true
  | Prim.u32le => fixedRead bs c 4 false
  | Prim.u64be =>
    match fixedRead bs c 4 true with
    | Except.ok (Val.num hi, c1) =>
      match fixedRead bs c1 4 true with
      | Except.ok (Val.num lo, c2) => Except.ok (Val.num (hi * 4294967296 + lo), c2)
      | Except.ok a => Except.error { offset := c1.pos, kind := Kind.invalid }
      | Except.error e => Except.error e
    | Except.ok a => Except.error { offset := c.pos, kind := Kind.invalid }
    | Except.error e => Except.error e
  | Prim.bytes n =>
    if need c n = true then Except.ok (Val.slice c.pos (c.pos + n.toNat), advance c n.toNat)
    else Except.error { offset := c.pos, kind := Kind.unexpectedEnd }
  | Prim.skip n =>
    if need c n = true then Except.ok (Val.unit, advance c n.toNat)
    else Except.error { offset := c.pos, kind := Kind.unexpectedEnd }
  | Prim.expect lit =>
    if (!need c ↑lit.length) = true then Except.error { offset := c.pos, kind := Kind.unexpectedEnd }
    else
      if ((List.range lit.length).all fun i => byteAt bs (c.pos + i) == lit.getD i 0) = true then
        Except.ok (Val.unit, advance c lit.length)
      else Except.error { offset := c.pos, kind := Kind.mismatch }
  | Prim.rest => Except.ok (Val.slice c.pos c.stop, { start := c.start, pos := c.stop, stop := c.stop })

def Codecs.readRange : Cursor → Cursor → Val → List (Nat × Nat) :=
fun c c' v =>
  match v with
  | Val.slice lo hi => [(c.pos, c'.pos), (lo, hi)]
  | x => [(c.pos, c'.pos)]
def Codecs.iterate : (Cursor → Except Fail R) → Nat → R → Except Fail R :=
fun f x x_1 => Nat.brecOn (motive := fun x => R → Except Fail R) x (iterate._f f) x_1
def Codecs.toEndLoop : (Cursor → Except Fail R) → Nat → Nat → Nat → R → Except Fail R :=
fun f max x x_1 x_2 => Nat.brecOn (motive := fun x => Nat → R → Except Fail R) x (toEndLoop._f f max) x_1 x_2
def Codecs.complete : (Cursor → Except Fail R) → Nat → Nat → Except Fail R :=
fun f lo hi =>
  match f { start := lo, pos := lo, stop := hi } with
  | Except.ok r =>
    if (r.c.pos == hi) = true then Except.ok r else Except.error { offset := r.c.pos, kind := Kind.trailing }
  | Except.error e => Except.error e
def Codecs.run : List Nat → G → Cursor → Except Fail R :=
fun bs x x_1 => G.brecOn (motive := fun x => Cursor → Except Fail R) x (run._f bs) x_1

def Codecs.header : List Nat → Nat → Nat → Bool → Except Fail (Nat × Nat × Nat) :=
fun bs pos hi top =>
  have size32 := be bs pos 4;
  have type := be bs (pos + 4) 4;
  have p := pos + 8;
  have sized :=
    if (size32 == 1) = true then
      if hi - p < 4 then Except.error { offset := p, kind := Kind.unexpectedEnd }
      else
        if hi - p < 8 then Except.error { offset := p + 4, kind := Kind.unexpectedEnd }
        else Except.ok (p + 8, be bs p 8)
    else
      if (size32 == 0) = true then
        if top = true then Except.ok (p, hi - p + 8) else Except.error { offset := p, kind := Kind.invalid }
      else Except.ok (p, size32);
  match sized with
  | Except.error e => Except.error e
  | Except.ok (p, size) =>
    if (type == uuid) = true then
      if hi - p < 16 then Except.error { offset := p, kind := Kind.unexpectedEnd } else Except.ok (type, p + 16, size)
    else Except.ok (type, p, size)
def Codecs.boxes : List Nat → Nat → Bool → Nat → Nat → Nat → List Box → Nat → Except Fail (List Box × Nat) :=
fun bs hi top maxBoxes x x_1 x_2 x_3 =>
  Nat.brecOn (motive := fun x => Nat → List Box → Nat → Except Fail (List Box × Nat)) x (boxes._f bs hi top maxBoxes)
    x_1 x_2 x_3
def Codecs.children : List Nat → Nat → Nat → Bool → Nat → Nat → Except Fail (List Box × Nat) :=
fun bs lo hi top count maxBoxes => boxes bs hi top maxBoxes (hi - lo + 1) lo [] count
def Codecs.bitAt : List Nat → Nat → Nat :=
fun bs i => if i < 8 * bs.length then byteAt bs (i / 8) / 2 ^ (7 - i % 8) % 2 else 0
def Codecs.bitsVal : List Nat → Nat → Nat → Nat :=
fun bs p n => List.foldl (fun acc i => acc * 2 + bitAt bs (p + i)) 0 (List.range n)

def Codecs.readBits : List Nat → Bool → Bits → Nat → Except Fail (Nat × Bits) :=
fun bs zeroFill r n =>
  if (n == 0) = true then Except.ok (0, r)
  else
    if r.pos + n ≤ 8 * bs.length then Except.ok (bitsVal bs r.pos n, { pos := r.pos + n, exhausted := r.exhausted })
    else
      if zeroFill = true then Except.ok (bitsVal bs r.pos n, { pos := 8 * bs.length, exhausted := true })
      else Except.error { offset := bs.length, kind := Kind.unexpectedEnd }
def Codecs.Good : Cursor → Prop :=
fun c => c.start ≤ c.pos ∧ c.pos ≤ c.stop
def Codecs.Within : Cursor → R → Prop :=
fun c r =>
  r.c.start = c.start ∧
    r.c.stop = c.stop ∧
      c.pos ≤ r.c.pos ∧
        r.c.pos ≤ c.stop ∧ ∀ (x : Nat × Nat), x ∈ r.reads → c.pos ≤ x.fst ∧ x.fst ≤ x.snd ∧ x.snd ≤ r.c.pos
def Codecs.BoxesWithin : Nat → Nat → List Box → Prop :=
fun lo hi bxs =>
  (∀ (b : Box), b ∈ bxs → lo ≤ b.offset ∧ b.offset + 8 ≤ b.lo ∧ b.lo ≤ b.hi ∧ b.hi ≤ hi) ∧
    List.Pairwise (fun a b => a.hi ≤ b.offset) bxs
