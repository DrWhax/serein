pad_length_bounds : ∀ (n : Nat),
  (paddedLength n + 1) % 80 = 0 ∧ n + 2 ≤ paddedLength n + 1 ∧ paddedLength n + 1 < n + 2 + 80
pad_length_iff : ∀ (m m' : List Nat), (pad m).length = (pad m').length ↔ (m.length + 1) / 80 = (m'.length + 1) / 80
unpad_pad : ∀ (m : List Nat), unpad (pad m) = some m
unpad_eq_some : ∀ (d m : List Nat), unpad d = some m ↔ ∃ k, d = m ++ 128 :: List.replicate k 0
unpad_upstream_differs : ∀ (d : List Nat), (∀ (x : Nat), x ∈ d → x = 0) → unpad d = none ∧ unpadUpstream d = some d
unpad_upstream_same : ∀ (d : List Nat), (¬∀ (x : Nat), x ∈ d → x = 0) → unpad d = unpadUpstream d
@dec_inj : ∀ {m n : Nat}, dec m = dec n → m = n
@b64_inj : ∀ {l l' : List Nat}, Bytes l → Bytes l' → b64 l = b64 l' → l = l'
separated : ∀ (u u' : Use), u.WF → u'.WF → (derivation u).canonical = (derivation u').canonical → u = u'
@key_binds : ∀ {k k' : Key}, k.WF → k'.WF → k.material = k'.material → k = k'
bucket_least : ∀ (L : List Nat),
  List.Pairwise (fun x1 x2 => x1 < x2) L →
    ∀ (n b : Nat), bucket L n = some b → b ∈ L ∧ n ≤ b ∧ ∀ (b' : Nat), b' ∈ L → n ≤ b' → b ≤ b'
lengths_sorted : ∀ (f : Field), List.Pairwise (fun x1 x2 => x1 < x2) f.lengths
padTo_length : ∀ (L item p : List Nat),
  padTo L item = some p →
    bucket L item.length = some p.length ∧
      item.length ≤ p.length ∧ p = item ++ List.replicate (p.length - item.length) 0
padTo_none : ∀ (L item : List Nat), padTo L item = none ↔ ∀ (b : Nat), b ∈ L → b < item.length
name_roundtrip : ∀ (g f p : List Nat),
  ¬0 ∈ g → ¬0 ∈ f → padTo Field.name.lengths (nameBytes g f) = some p → decodeName p = (g, f)
string_roundtrip : ∀ (L s p : List Nat), ¬0 ∈ s → padTo L s = some p → decodeString p = s
name_nul_splits : ∃ g p,
  padTo Field.name.lengths (nameBytes g []) = some p ∧ decodeName p = ([65], [66]) ∧ g = [65, 0, 66]
manifest_ok : ∀ (c : ManifestCase) (v : Nat),
  verifyManifest c = Except.ok v →
    v = c.version ∧
      fresh c.current c.version = true ∧
        ∃ k m,
          c.origin = Origin.sealed k m ∧
            k.material = (manifestKey c.storage c.version).material ∧ m.parses = true ∧ m.inner = c.version
manifest_bound : ∀ (c : ManifestCase) (v : Nat),
  verifyManifest c = Except.ok v →
    ∀ (k : Key) (m : ManifestContent), c.origin = Origin.sealed k m → k.WF → k = manifestKey c.storage c.version
items_ok : ∀ (c : ItemsCase) (ids : List (List Nat)) (n : Nat),
  verifyItems c = Except.ok (ids, n) →
    ids = List.map (fun x => x.fst) c.items ∧
      ∀ (i : List Nat × Origin RecordContent),
        i ∈ c.items →
          i.fst ∈ c.requested ∧
            ∃ k r, i.snd = Origin.sealed k r ∧ k.material = (itemKey c.storage c.ikm i.fst).material ∧ r.parses = true
items_bound : ∀ (c : ItemsCase) (ids : List (List Nat)) (n : Nat),
  verifyItems c = Except.ok (ids, n) →
    ∀ (i : List Nat × Origin RecordContent),
      i ∈ c.items →
        Bytes i.fst →
          ∀ (k : Key) (r : RecordContent),
            i.snd = Origin.sealed k r → k.WF → i.fst ∈ c.requested ∧ k = itemKey c.storage c.ikm i.fst

  decryptProfile c = Except.ok v →
    29 ≤ c.len ∧ ∃ p, c.origin = Origin.sealed (Key.profile c.key) p ∧ v = readField c.field p.plain
