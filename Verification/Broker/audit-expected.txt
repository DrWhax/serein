def Broker.accept : Kind → Reply → Status × Option Nat :=
fun x x_1 =>
  match x, x_1 with
  | Kind.thumb size input, Reply.thumb w h =>
    (if w = size ∧ h = size then Status.ok (Val.thumb w h) else Status.failed Err.refused, none)
  | Kind.openV input, Reply.opened u true => (Status.ok (Val.session u), none)
  | Kind.openV input, Reply.opened u false => (Status.failed Err.refused, some u)
  | Kind.start, Reply.voiceId u => (Status.ok (Val.session u), none)
  | Kind.release c u, Reply.empty => (Status.ok Val.empty, none)
  | x, Reply.error => (Status.failed Err.helper, none)
  | x, x_2 => (Status.failed Err.refused, none)
def Broker.late : Kind → Reply → Option Nat :=
fun x x_1 =>
  match x, x_1 with
  | Kind.openV input, Reply.opened u valid => some u
  | Kind.start, Reply.voiceId u => some u
  | x, x_2 => none
def Broker.ended : St → Nat → Bool :=
fun s id => (Option.map (fun x => x.ended) s.sess[id]?).getD true
def Broker.opened : St → Ch → St :=
fun s c =>
  match s.live c with
  | some val => s
  | none =>
    { reqs := s.reqs, sess := s.sess ++ [{ ch := c, ended := false }],
      live := fun c' => if c' = c then some s.sess.length else s.live c', strikes := s.strikes,
      delivered := s.delivered }
def Broker.send : St → Kind → St :=
fun s k =>
  have __src := opened s k.ch;
  {
    reqs :=
      (opened s k.ch).reqs ++
        [{ kind := k, sess := (opened s k.ch).live k.ch, status := Status.pending, answered := false }],
    sess := __src.sess, live := __src.live, strikes := __src.strikes, delivered := __src.delivered }
def Broker.quarantined : St → Kind → Bool :=
fun s k =>
  match k.input with
  | some i => decide (quarantineAt ≤ s.strikes i)
  | none => false
def Broker.call : St → Kind → St :=
fun s k =>
  match k with
  | Kind.release c u => s
  | x =>
    if quarantined s k = true then
      { reqs := s.reqs ++ [{ kind := k, sess := none, status := Status.failed Err.quarantined, answered := true }],
        sess := s.sess, live := s.live, strikes := s.strikes, delivered := s.delivered }
    else send s k
def Broker.answer : Req → Reply → Status × Option Nat :=
fun q r => if q.status = Status.pending then accept q.kind r else (q.status, late q.kind r)
def Broker.reply : St → Nat → Reply → St :=
fun s m r =>
  match s.reqs[m]? with
  | none => s
  | some q =>
    match q.sess with
    | none => s
    | some id =>
      if q.answered = true ∨ ended s id = true then s
      else
        have s1 :=
          { reqs := s.reqs.set m { kind := q.kind, sess := q.sess, status := (answer q r).fst, answered := true },
            sess := s.sess, live := s.live, strikes := s.strikes, delivered := s.delivered ++ [(m, r)] };
        match (answer q r).snd with
        | some u => send s1 (Kind.release q.kind.ch u)
        | none => s1
def Broker.endReq : Nat → Req → Req :=
fun id q =>
  if q.sess = some id ∧ q.answered = false then
    { kind := q.kind, sess := q.sess, status := if q.status = Status.pending then Status.failed Err.lost else q.status,
      answered := true }
  else q
def Broker.inFlight : St → Nat → Nat → Bool :=
fun s id i => s.reqs.any fun q => decide (q.sess = some id ∧ q.status = Status.pending ∧ q.kind.input = some i)
def Broker.crash : St → Ch → St :=
fun s c =>
  match s.live c with
  | none => s
  | some id =>
    { reqs := List.map (endReq id) s.reqs, sess := s.sess.set id { ch := c, ended := true },
      live := fun c' => if c' = c then none else s.live c',
      strikes := fun i => s.strikes i + if inFlight s id i = true then 1 else 0, delivered := s.delivered }
def Broker.bump : (Nat → Nat) → Option Nat → Nat → Nat → Nat :=
fun f i n j => if i = some j then f j + n else f j
def Broker.timeout : St → Nat → St :=
fun s m =>
  match s.reqs[m]? with
  | none => s
  | some q =>
    if q.status = Status.pending then
      {
        reqs :=
          s.reqs.set m { kind := q.kind, sess := q.sess, status := Status.failed Err.timedOut, answered := q.answered },
        sess := s.sess, live := s.live, strikes := bump s.strikes q.kind.input quarantineAt, delivered := s.delivered }
    else s
def Broker.cancel : St → Nat → St :=
fun s m =>
  match s.reqs[m]? with
  | none => s
  | some q =>
    if q.status = Status.pending then
      {
        reqs :=
          s.reqs.set m
            { kind := q.kind, sess := q.sess, status := Status.failed Err.cancelled, answered := q.answered },
        sess := s.sess, live := s.live, strikes := s.strikes, delivered := s.delivered }
    else s
def Broker.step : St → Ev → St :=
fun s x =>
  match x with
  | Ev.call k => call s k
  | Ev.reply m r => reply s m r
  | Ev.crash c => crash s c
  | Ev.unsolicited c => s
  | Ev.timeout m => timeout s m
  | Ev.cancel m => cancel s m

def Broker.Shaped : St → Prop :=
fun s => ∀ (q : Req), q ∈ s.reqs → ∀ (id : Nat), q.sess = some id → ∃ x, s.sess[id]? = some x ∧ x.ch = q.kind.ch
def Broker.LiveOpen : St → Prop :=
fun s => ∀ (c : Ch) (id : Nat), s.live c = some id → s.sess[id]? = some { ch := c, ended := false }
def Broker.OneOpen : St → Prop :=
fun s => ∀ (id : Nat) (x : Sess), s.sess[id]? = some x → x.ended = false → s.live x.ch = some id
def Broker.PendingLive : St → Prop :=
fun s =>
  ∀ (q : Req), q ∈ s.reqs → q.status = Status.pending → q.answered = false ∧ ∃ id, q.sess = some id ∧ ended s id = false
def Broker.LostEnded : St → Prop :=
fun s =>
  ∀ (q : Req),
    q ∈ s.reqs → q.status = Status.failed Err.lost → ∃ id x, q.sess = some id ∧ s.sess[id]? = some x ∧ x.ended = true
def Broker.Answered : St → Prop :=
fun s => ∀ (m : Nat) (r : Reply), (m, r) ∈ s.delivered → ∃ q, s.reqs[m]? = some q ∧ q.answered = true
def Broker.OkOwn : St → Prop :=
fun s =>
  ∀ (m : Nat) (q : Req) (v : Val),
    s.reqs[m]? = some q → q.status = Status.ok v → ∃ r, (m, r) ∈ s.delivered ∧ (accept q.kind r).fst = Status.ok v
def Broker.NoLeak : St → Prop :=
fun s =>
  ∀ (m : Nat) (r : Reply) (q : Req) (u : Nat),
    (m, r) ∈ s.delivered →
      s.reqs[m]? = some q →
        late q.kind r = some u →
          q.status = Status.ok (Val.session u) ∨ Kind.release q.kind.ch u ∈ List.map Req.kind s.reqs
def Broker.ReleaseNamed : St → Prop :=
fun s =>
  ∀ (j : Nat) (q : Req) (c : Ch) (u : Nat),
    s.reqs[j]? = some q →
      q.kind = Kind.release c u →
        ∃ m r p, (m, r) ∈ s.delivered ∧ s.reqs[m]? = some p ∧ p.kind.ch = c ∧ late p.kind r = some u
def Broker.timedOutCount : St → Nat → Nat :=
fun s i => List.countP (fun q => decide (q.kind.input = some i ∧ q.status = Status.failed Err.timedOut)) s.reqs
def Broker.lostOn : St → Nat → Nat → Bool :=
fun s i id => s.reqs.any fun q => decide (q.sess = some id ∧ q.kind.input = some i ∧ q.status = Status.failed Err.lost)
def Broker.lostSessions : St → Nat → Nat :=
fun s i => List.countP (lostOn s i) (List.range s.sess.length)
def Broker.StrikesExact : St → Prop :=
fun s => ∀ (i : Nat), s.strikes i = quarantineAt * timedOutCount s i + lostSessions s i
def Broker.Inv : St → Prop :=
fun s =>
  Shaped s ∧
    LiveOpen s ∧
      OneOpen s ∧ PendingLive s ∧ LostEnded s ∧ Answered s ∧ OkOwn s ∧ NoLeak s ∧ ReleaseNamed s ∧ StrikesExact s
