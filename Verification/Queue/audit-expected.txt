def Queue.start : S → S :=
fun s =>
  if (s.draining != 0) = true then s
  else
    if s.loop.isSome = true then
      { db := s.db, wantsRunning := true, loop := s.loop, cancelled := s.cancelled, draining := s.draining,
        wakes := s.wakes, stopWaiters := s.stopWaiters, returns := s.returns, stopNext := s.stopNext, done := s.done,
        returned := s.returned }
    else
      { db := s.db, wantsRunning := true, loop := some Phase.select, cancelled := false, draining := s.draining,
        wakes := s.wakes, stopWaiters := s.stopWaiters, returns := s.returns, stopNext := s.stopNext, done := s.done,
        returned := s.returned }
def Queue.stop : S → S :=
fun s =>
  { db := s.db, wantsRunning := false, loop := s.loop, cancelled := s.cancelled || s.loop.isSome,
    draining := s.draining, wakes := s.wakes + 1, stopWaiters := s.stopWaiters, returns := s.returns,
    stopNext := s.stopNext, done := s.done, returned := s.returned }
def Queue.stopAndWait : S → S :=
fun s =>
  have n := s.stopNext;
  have s :=
    stop
      { db := s.db, wantsRunning := s.wantsRunning, loop := s.loop, cancelled := s.cancelled,
        draining := s.draining + 1, wakes := s.wakes, stopWaiters := s.stopWaiters, returns := s.returns,
        stopNext := n + 1, done := s.done, returned := s.returned };
  if s.loop.isSome = true then
    { db := s.db, wantsRunning := s.wantsRunning, loop := s.loop, cancelled := s.cancelled, draining := s.draining,
      wakes := s.wakes, stopWaiters := s.stopWaiters ++ [n], returns := s.returns, stopNext := s.stopNext,
      done := s.done, returned := s.returned }
  else
    { db := s.db, wantsRunning := s.wantsRunning, loop := s.loop, cancelled := s.cancelled, draining := s.draining,
      wakes := s.wakes, stopWaiters := s.stopWaiters, returns := s.returns ++ [n], stopNext := s.stopNext,
      done := s.done, returned := s.returned }
def Queue.loopEnds : S → S :=
fun s =>
  have t :=
    { db := s.db, wantsRunning := s.wantsRunning, loop := none, cancelled := false, draining := s.draining,
      wakes := s.wakes, stopWaiters := [], returns := s.returns ++ s.stopWaiters, stopNext := s.stopNext,
      done := s.done, returned := s.returned };
  if s.wantsRunning = true then start t else t
def Queue.loopStep : S → S :=
fun s =>
  match s.loop with
  | some Phase.select =>
    if s.cancelled = true then loopEnds s
    else
      match s.db.head? with
      | some j =>
        { db := s.db, wantsRunning := s.wantsRunning, loop := some (Phase.executing j), cancelled := s.cancelled,
          draining := s.draining, wakes := s.wakes, stopWaiters := s.stopWaiters, returns := s.returns,
          stopNext := s.stopNext, done := s.done, returned := s.returned }
      | none =>
        { db := s.db, wantsRunning := s.wantsRunning, loop := some Phase.waiting, cancelled := s.cancelled,
          draining := s.draining, wakes := s.wakes, stopWaiters := s.stopWaiters, returns := s.returns,
          stopNext := s.stopNext, done := s.done, returned := s.returned }
  | some (Phase.after j o) =>
    have o := if (s.cancelled && o != Outcome.ok) = true then Outcome.cancelled else o;
    have db :=
      match o with
      | Outcome.ok => s.db.erase j
      | Outcome.permanent => s.db.erase j
      | Outcome.retry => s.db
      | Outcome.cancelled => s.db;
    have done :=
      match o with
      | Outcome.ok => s.done ++ [j]
      | Outcome.permanent => s.done ++ [j]
      | x => s.done;
    { db := db, wantsRunning := s.wantsRunning, loop := some Phase.select, cancelled := s.cancelled,
      draining := s.draining, wakes := s.wakes, stopWaiters := s.stopWaiters, returns := s.returns,
      stopNext := s.stopNext, done := done, returned := s.returned }
  | x => s
def Queue.step : S → Ev → S :=
fun s x =>
  match x with
  | Ev.runLoop =>
    match s.loop with
    | some Phase.select => loopStep s
    | some (Phase.after j o) => loopStep s
    | x => s
  | Ev.runWake =>
    if s.wakes = 0 then s
    else
      have s :=
        { db := s.db, wantsRunning := s.wantsRunning, loop := s.loop, cancelled := s.cancelled, draining := s.draining,
          wakes := s.wakes - 1, stopWaiters := s.stopWaiters, returns := s.returns, stopNext := s.stopNext,
          done := s.done, returned := s.returned };
      match s.loop with
      | some Phase.waiting =>
        { db := s.db, wantsRunning := s.wantsRunning, loop := some Phase.select, cancelled := s.cancelled,
          draining := s.draining, wakes := s.wakes, stopWaiters := s.stopWaiters, returns := s.returns,
          stopNext := s.stopNext, done := s.done, returned := s.returned }
      | x => s
  | Ev.runReturn k =>
    match s.returns[k]? with
    | some n =>
      { db := s.db, wantsRunning := s.wantsRunning, loop := s.loop, cancelled := s.cancelled,
        draining := s.draining - 1, wakes := s.wakes, stopWaiters := s.stopWaiters, returns := s.returns.eraseIdx k,
        stopNext := s.stopNext, done := s.done, returned := s.returned ++ [n] }
    | none => s
  | Ev.start => start s
  | Ev.stop => stop s
  | Ev.stopAndWait => stopAndWait s
  | Ev.enqueue j =>
    { db := s.db ++ [j], wantsRunning := s.wantsRunning, loop := s.loop, cancelled := s.cancelled,
      draining := s.draining, wakes := s.wakes + 1, stopWaiters := s.stopWaiters, returns := s.returns,
      stopNext := s.stopNext, done := s.done, returned := s.returned }
  | Ev.handlerReturns o =>
    match s.loop with
    | some (Phase.executing j) =>
      { db := s.db, wantsRunning := s.wantsRunning, loop := some (Phase.after j o), cancelled := s.cancelled,
        draining := s.draining, wakes := s.wakes, stopWaiters := s.stopWaiters, returns := s.returns,
        stopNext := s.stopNext, done := s.done, returned := s.returned }
    | x => s

def Queue.shutdownClean : S → Ev → S → Bool :=
fun s _e s' => s'.returned.length == s.returned.length || s'.loop == none
def Queue.noRestartWhileDraining : S → Ev → S → Bool :=
fun s e s' =>
  match e with
  | Ev.start => s.draining == 0 || s' == s
  | x => true
def Queue.oneAtATime : S → Ev → S → Bool :=
fun s e s' =>
  decide (s'.done.length ≤ s.done.length + 1) &&
    (s'.done.length == s.done.length ||
      match e, s.loop with
      | Ev.runLoop, some (Phase.after j o) => s'.done == s.done ++ [j]
      | x, x_1 => false)
def Queue.measure : S → Nat :=
fun s => 4 * weight s.loop + s.wakes + s.returns.length + s.stopWaiters.length
@[reducible] def measure.{u} : {α : Sort u} → (α → Nat) → WellFoundedRelation α :=
fun {α} f => invImage f Nat.lt_wfRel
def Queue.Ev.isDrain : Ev → Bool :=
fun x =>
  match x with
  | Ev.runLoop => true
  | Ev.runWake => true
  | Ev.runReturn k => true
  | Ev.handlerReturns o => true
  | x => false
