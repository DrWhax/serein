def CallBoundary.St.init : St :=
{ ep := some 0, eps := 1, epGen := [0], gen := 0, reported := 0, gone := [], config := false, configOn := [],
  selfKnown := false, call := none, offers := [], groups := [], held := [], senders := [], cancelled := [], out := [],
  cmds := [] }
def CallBoundary.selfId : Nat :=
0
def CallBoundary.send : St → Origin → Option Nat → St × Bool :=
fun s o g =>
  match g with
  | some g' => if g' = s.gen then send.sendLive s o else (s, false)
  | none => send.sendLive s o
def CallBoundary.send.sendLive : St → Origin → St × Bool :=
fun s o =>
  match s.ep with
  | some e =>
    if e ∈ s.gone then
      ({ ep := none, eps := s.eps, epGen := s.epGen, gen := s.gen + 1, reported := s.reported, gone := s.gone,
          config := s.config, configOn := s.configOn, selfKnown := s.selfKnown, call := s.call, offers := s.offers,
          groups := s.groups, held := s.held, senders := s.senders, cancelled := s.cancelled, out := s.out,
          cmds := s.cmds },
        false)
    else
      ({ ep := s.ep, eps := s.eps, epGen := s.epGen, gen := s.gen, reported := s.reported, gone := s.gone,
          config := s.config, configOn := s.configOn, selfKnown := s.selfKnown, call := s.call, offers := s.offers,
          groups := s.groups, held := s.held, senders := s.senders, cancelled := s.cancelled, out := s.out,
          cmds := s.cmds ++ [{ ep := e, origin := o, gen := s.gen }] },
        true)
  | none =>
    have e := s.eps;
    have replay := if s.config = true then [{ ep := e, origin := Origin.rtc, gen := s.gen }] else [];
    ({ ep := some e, eps := e + 1, epGen := s.epGen ++ [s.gen], gen := s.gen, reported := s.reported, gone := s.gone,
        config := s.config, configOn := if s.config = true then s.configOn ++ [e] else s.configOn,
        selfKnown := s.selfKnown, call := s.call, offers := s.offers, groups := s.groups, held := s.held,
        senders := s.senders, cancelled := s.cancelled, out := s.out,
        cmds := s.cmds ++ replay ++ [{ ep := e, origin := o, gen := s.gen }] },
      true)
def CallBoundary.run : St → Origin → Option Nat → St :=
fun s o g => (send s o g).fst
def CallBoundary.lookup : List (Nat × Nat) → Nat → Option Nat :=
fun offers callId => Option.map (fun x => x.snd) (List.find? (fun x => decide (x.fst = callId)) offers)
def CallBoundary.findGroup : St → Nat → Option GroupR :=
fun s client => List.find? (fun x => decide (x.client = client)) s.groups
def CallBoundary.findHeld : St → Nat → Option GroupR :=
fun s client => List.find? (fun x => decide (x.client = client)) s.held
def CallBoundary.insertGroup : GroupR → List GroupR → List GroupR :=
fun g x => List.brecOn x (insertGroup._f g)
def CallBoundary.updGroup : Nat → (GroupR → GroupR) → List GroupR → List GroupR :=
fun client f gs => List.map (fun g => if g.client = client then f g else g) gs
def CallBoundary.live : St → Bool :=
fun s =>
  match s.ep with
  | some e => !s.gone.contains e
  | none => false
def CallBoundary.maySignal : St → Nat → Nat → Bool :=
fun s peer callId =>
  (match s.call with
    | some c => c.peer == peer && c.id == callId
    | none => false) ||
    lookup s.offers callId == some peer
def CallBoundary.mayMessage : St → Nat → Bool :=
fun s to =>
  s.selfKnown && to == selfId ||
    s.groups.any fun g => !g.link && g.members.contains to || g.link && g.participants.contains to
def CallBoundary.mayGroupMessage : St → Nat → List Nat → Bool :=
fun s gid to => ((s.groups.any fun g => !g.link && g.group == gid) || s.cancelled.contains gid) && to.all (mayMessage s)
def CallBoundary.report : St → St :=
fun s =>
  have old := List.filter (fun x => decide (x.gen < s.gen)) s.groups;
  { ep := s.ep, eps := s.eps, epGen := s.epGen, gen := s.gen, reported := s.gen, gone := s.gone, config := s.config,
    configOn := s.configOn, selfKnown := s.selfKnown,
    call :=
      Option.map (fun c => if c.gen < s.gen then { peer := c.peer, id := c.id, gen := c.gen, ended := true } else c)
        s.call,
    offers :=
      match s.call with
      | some c =>
        if (decide (c.gen < s.gen) && !c.ended) = true then List.filter (fun x => decide (x.fst ≠ c.id)) s.offers
        else s.offers
      | none => s.offers,
    groups := List.filter (fun x => decide (x.gen ≥ s.gen)) s.groups, held := s.held, senders := s.senders,
    cancelled := s.cancelled, out := s.out ++ List.map (fun x => Out.groupEnded x.client) old, cmds := s.cmds }
def CallBoundary.step : St → Ev → St :=
fun s x =>
  match x with
  | Ev.setSelf =>
    have s1 :=
      { ep := s.ep, eps := s.eps, epGen := s.epGen, gen := s.gen, reported := s.reported, gone := s.gone,
        config := true, configOn := s.configOn, selfKnown := true, call := s.call, offers := s.offers,
        groups := s.groups, held := s.held, senders := s.senders, cancelled := s.cancelled, out := s.out,
        cmds := s.cmds };
    match send s1 Origin.rtc none with
    | (s2, true) =>
      match s2.ep with
      | some e =>
        if e ∈ s2.configOn then s2
        else
          { ep := s2.ep, eps := s2.eps, epGen := s2.epGen, gen := s2.gen, reported := s2.reported, gone := s2.gone,
            config := s2.config, configOn := s2.configOn ++ [e], selfKnown := s2.selfKnown, call := s2.call,
            offers := s2.offers, groups := s2.groups, held := s2.held, senders := s2.senders, cancelled := s2.cancelled,
            out := s2.out, cmds := s2.cmds }
      | none => s2
    | (s2, false) => s2
  | Ev.start peer callId =>
    match send s Origin.rtc none with
    | (s1, true) =>
      { ep := s1.ep, eps := s1.eps, epGen := s1.epGen, gen := s1.gen, reported := s1.reported, gone := s1.gone,
        config := s1.config, configOn := s1.configOn, selfKnown := s1.selfKnown,
        call := some { peer := peer, id := callId, gen := s1.gen, ended := false }, offers := s1.offers,
        groups := s1.groups, held := s1.held, senders := s1.senders, cancelled := s1.cancelled, out := s1.out,
        cmds := s1.cmds }
    | (s1, false) => s1
  | Ev.offer peer callId =>
    run
      { ep := s.ep, eps := s.eps, epGen := s.epGen, gen := s.gen, reported := s.reported, gone := s.gone,
        config := s.config, configOn := s.configOn, selfKnown := s.selfKnown, call := s.call,
        offers := (callId, peer) :: List.filter (fun x => decide (x.fst ≠ callId)) s.offers, groups := s.groups,
        held := s.held, senders := s.senders, cancelled := s.cancelled, out := s.out, cmds := s.cmds }
      Origin.rtc none
  | Ev.callMsg u =>
    run
      { ep := s.ep, eps := s.eps, epGen := s.epGen, gen := s.gen, reported := s.reported, gone := s.gone,
        config := s.config, configOn := s.configOn, selfKnown := s.selfKnown, call := s.call, offers := s.offers,
        groups := s.groups, held := s.held, senders := u :: s.senders.erase u, cancelled := s.cancelled, out := s.out,
        cmds := s.cmds }
      Origin.rtc none
  | Ev.group gid client link =>
    match send s Origin.rtc none with
    | (s1, true) =>
      if client = 0 then s1
      else
        { ep := s1.ep, eps := s1.eps, epGen := s1.epGen, gen := s1.gen, reported := s1.reported, gone := s1.gone,
          config := s1.config, configOn := s1.configOn, selfKnown := s1.selfKnown, call := s1.call, offers := s1.offers,
          groups :=
            insertGroup
              { client := client, group := gid, link := link, gen := s1.gen, members := [], participants := [] }
              s1.groups,
          held :=
            insertGroup
              { client := client, group := gid, link := link, gen := s1.gen, members := [], participants := [] }
              s1.held,
          senders := s1.senders, cancelled := s1.cancelled, out := s1.out, cmds := s1.cmds }
    | (s1, false) => s1
  | Ev.members client ms =>
    match findGroup s client with
    | some g =>
      run
        { ep := s.ep, eps := s.eps, epGen := s.epGen, gen := s.gen, reported := s.reported, gone := s.gone,
          config := s.config, configOn := s.configOn, selfKnown := s.selfKnown, call := s.call, offers := s.offers,
          groups :=
            updGroup client
              (fun g =>
                { client := g.client, group := g.group, link := g.link, gen := g.gen, members := ms,
                  participants := g.participants })
              s.groups,
          held := s.held, senders := s.senders, cancelled := s.cancelled, out := s.out, cmds := s.cmds }
        Origin.group (some g.gen)
    | none => s
  | Ev.callCmd =>
    match s.call with
    | some c => run s Origin.call (some c.gen)
    | none => s
  | Ev.groupCmd client =>
    match findHeld s client with
    | some g => run s Origin.group (some g.gen)
    | none => s
  | Ev.cancelRing gid =>
    run
      { ep := s.ep, eps := s.eps, epGen := s.epGen, gen := s.gen, reported := s.reported, gone := s.gone,
        config := s.config, configOn := s.configOn, selfKnown := s.selfKnown, call := s.call, offers := s.offers,
        groups := s.groups, held := s.held, senders := s.senders, cancelled := gid :: s.cancelled.erase gid,
        out := s.out, cmds := s.cmds }
      Origin.rtc none
  | Ev.expireSender u =>
    { ep := s.ep, eps := s.eps, epGen := s.epGen, gen := s.gen, reported := s.reported, gone := s.gone,
      config := s.config, configOn := s.configOn, selfKnown := s.selfKnown, call := s.call, offers := s.offers,
      groups := s.groups, held := s.held, senders := s.senders.erase u, cancelled := s.cancelled, out := s.out,
      cmds := s.cmds }
  | Ev.expireRing gid =>
    { ep := s.ep, eps := s.eps, epGen := s.epGen, gen := s.gen, reported := s.reported, gone := s.gone,
      config := s.config, configOn := s.configOn, selfKnown := s.selfKnown, call := s.call, offers := s.offers,
      groups := s.groups, held := s.held, senders := s.senders, cancelled := s.cancelled.erase gid, out := s.out,
      cmds := s.cmds }
  | Ev.incoming peer callId =>
    if (live s && lookup s.offers callId == some peer) = true then
      { ep := s.ep, eps := s.eps, epGen := s.epGen, gen := s.gen, reported := s.reported, gone := s.gone,
        config := s.config, configOn := s.configOn, selfKnown := s.selfKnown,
        call := some { peer := peer, id := callId, gen := s.gen, ended := false }, offers := s.offers,
        groups := s.groups, held := s.held, senders := s.senders, cancelled := s.cancelled,
        out := s.out ++ [Out.incoming peer callId], cmds := s.cmds }
    else s
  | Ev.signal peer callId =>
    if (!live s) = true then s
    else
      if maySignal s peer callId = true then
        run
          { ep := s.ep, eps := s.eps, epGen := s.epGen, gen := s.gen, reported := s.reported, gone := s.gone,
            config := s.config, configOn := s.configOn, selfKnown := s.selfKnown, call := s.call, offers := s.offers,
            groups := s.groups, held := s.held, senders := s.senders, cancelled := s.cancelled,
            out := s.out ++ [Out.signal peer callId], cmds := s.cmds }
          Origin.rtc none
      else run s Origin.rtc none
  | Ev.message to =>
    if (live s && mayMessage s to) = true then
      { ep := s.ep, eps := s.eps, epGen := s.epGen, gen := s.gen, reported := s.reported, gone := s.gone,
        config := s.config, configOn := s.configOn, selfKnown := s.selfKnown, call := s.call, offers := s.offers,
        groups := s.groups, held := s.held, senders := s.senders, cancelled := s.cancelled,
        out := s.out ++ [Out.message to], cmds := s.cmds }
    else s
  | Ev.groupMessage gid to =>
    if (live s && mayGroupMessage s gid to) = true then
      { ep := s.ep, eps := s.eps, epGen := s.epGen, gen := s.gen, reported := s.reported, gone := s.gone,
        config := s.config, configOn := s.configOn, selfKnown := s.selfKnown, call := s.call, offers := s.offers,
        groups := s.groups, held := s.held, senders := s.senders, cancelled := s.cancelled,
        out := s.out ++ [Out.groupMessage gid to], cmds := s.cmds }
    else s
  | Ev.ring u gid =>
    if (live s && s.senders.contains u) = true then
      { ep := s.ep, eps := s.eps, epGen := s.epGen, gen := s.gen, reported := s.reported, gone := s.gone,
        config := s.config, configOn := s.configOn, selfKnown := s.selfKnown, call := s.call, offers := s.offers,
        groups := s.groups, held := s.held, senders := s.senders, cancelled := s.cancelled,
        out := s.out ++ [Out.ring u gid], cmds := s.cmds }
    else s
  | Ev.callEnded callId =>
    if (!live s) = true then s
    else
      have s1 :=
        { ep := s.ep, eps := s.eps, epGen := s.epGen, gen := s.gen, reported := s.reported, gone := s.gone,
          config := s.config, configOn := s.configOn, selfKnown := s.selfKnown, call := s.call,
          offers := List.filter (fun x => decide (x.fst ≠ callId)) s.offers, groups := s.groups, held := s.held,
          senders := s.senders, cancelled := s.cancelled, out := s.out, cmds := s.cmds };
      match s1.call with
      | some c =>
        if (decide (c.id = callId) && !c.ended) = true then
          have s2 :=
            { ep := s1.ep, eps := s1.eps, epGen := s1.epGen, gen := s1.gen, reported := s1.reported, gone := s1.gone,
              config := s1.config, configOn := s1.configOn, selfKnown := s1.selfKnown,
              call := some { peer := c.peer, id := c.id, gen := c.gen, ended := true }, offers := s1.offers,
              groups := s1.groups, held := s1.held, senders := s1.senders, cancelled := s1.cancelled, out := s1.out,
              cmds := s1.cmds };
          run (run (run s2 Origin.call (some c.gen)) Origin.call (some c.gen)) Origin.call (some c.gen)
        else s1
      | none => s1
  | Ev.devices client us =>
    if live s = true then
      { ep := s.ep, eps := s.eps, epGen := s.epGen, gen := s.gen, reported := s.reported, gone := s.gone,
        config := s.config, configOn := s.configOn, selfKnown := s.selfKnown, call := s.call, offers := s.offers,
        groups :=
          updGroup client
            (fun g =>
              { client := g.client, group := g.group, link := g.link, gen := g.gen, members := g.members,
                participants := us })
            s.groups,
        held := s.held, senders := s.senders, cancelled := s.cancelled, out := s.out, cmds := s.cmds }
    else s
  | Ev.groupEnded client =>
    if (!live s) = true then s
    else
      match findGroup s client with
      | some g =>
        have s1 :=
          run
            { ep := s.ep, eps := s.eps, epGen := s.epGen, gen := s.gen, reported := s.reported, gone := s.gone,
              config := s.config, configOn := s.configOn, selfKnown := s.selfKnown, call := s.call, offers := s.offers,
              groups := List.filter (fun x => decide (x.client ≠ client)) s.groups, held := s.held,
              senders := s.senders, cancelled := s.cancelled, out := s.out, cmds := s.cmds }
            Origin.group (some g.gen);
        { ep := s1.ep, eps := s1.eps, epGen := s1.epGen, gen := s1.gen, reported := s1.reported, gone := s1.gone,
          config := s1.config, configOn := s1.configOn, selfKnown := s1.selfKnown, call := s1.call, offers := s1.offers,
          groups := s1.groups, held := s1.held, senders := s1.senders, cancelled := s1.cancelled,
          out := s1.out ++ [Out.groupEnded client], cmds := s1.cmds }
      | none => s
  | Ev.crash =>
    match s.ep with
    | some e =>
      if e ∈ s.gone then s
      else
        { ep := s.ep, eps := s.eps, epGen := s.epGen, gen := s.gen, reported := s.reported, gone := s.gone ++ [e],
          config := s.config, configOn := s.configOn, selfKnown := s.selfKnown, call := s.call, offers := s.offers,
          groups := s.groups, held := s.held, senders := s.senders, cancelled := s.cancelled, out := s.out,
          cmds := s.cmds }
    | none => s
  | Ev.wait =>
    if s.reported < s.gen then report s
    else
      match s.ep with
      | some e =>
        if e ∈ s.gone then
          { ep := none, eps := s.eps, epGen := s.epGen, gen := s.gen + 1, reported := s.reported, gone := s.gone,
            config := s.config, configOn := s.configOn, selfKnown := s.selfKnown, call := s.call, offers := s.offers,
            groups := s.groups, held := s.held, senders := s.senders, cancelled := s.cancelled, out := s.out,
            cmds := s.cmds }
        else s
      | none => s
def CallBoundary.St.ran : St → Nat → Origin → St :=
fun s e o =>
  { ep := s.ep, eps := s.eps, epGen := s.epGen, gen := s.gen, reported := s.reported, gone := s.gone,
    config := s.config, configOn := s.configOn, selfKnown := s.selfKnown, call := s.call, offers := s.offers,
    groups := s.groups, held := s.held, senders := s.senders, cancelled := s.cancelled, out := s.out,
    cmds := s.cmds ++ [{ ep := e, origin := o, gen := s.gen }] }
def CallBoundary.St.fresh : St → Origin → St :=
fun s o =>
  { ep := some s.eps, eps := s.eps + 1, epGen := s.epGen ++ [s.gen], gen := s.gen, reported := s.reported,
    gone := s.gone, config := s.config, configOn := if s.config = true then s.configOn ++ [s.eps] else s.configOn,
    selfKnown := s.selfKnown, call := s.call, offers := s.offers, groups := s.groups, held := s.held,
    senders := s.senders, cancelled := s.cancelled, out := s.out,
    cmds :=
      (s.cmds ++ if s.config = true then [{ ep := s.eps, origin := Origin.rtc, gen := s.gen }] else []) ++
        [{ ep := s.eps, origin := o, gen := s.gen }] }
def CallBoundary.Allowed : St → Out → Prop :=
fun s x =>
  match x with
  | Out.signal p c => (∃ k, s.call = some k ∧ k.peer = p ∧ k.id = c) ∨ lookup s.offers c = some p
  | Out.incoming p c => lookup s.offers c = some p
  | Out.message u =>
    s.selfKnown = true ∧ u = selfId ∨
      ∃ g, g ∈ s.groups ∧ (g.link = false ∧ u ∈ g.members ∨ g.link = true ∧ u ∈ g.participants)
  | Out.groupMessage gid us =>
    ((∃ g, g ∈ s.groups ∧ g.link = false ∧ g.group = gid) ∨ gid ∈ s.cancelled) ∧
      ∀ (u : Nat),
        u ∈ us →
          s.selfKnown = true ∧ u = selfId ∨
            ∃ g, g ∈ s.groups ∧ (g.link = false ∧ u ∈ g.members ∨ g.link = true ∧ u ∈ g.participants)
  | Out.ring u group => u ∈ s.senders
  | Out.groupEnded c => ∃ g, g ∈ s.groups ∧ g.client = c
structure CallBoundary.Inv (s : St) : Prop
number of parameters: 1
fields:
  CallBoundary.Inv.epGenLen : s.epGen.length = s.eps
  CallBoundary.Inv.epLive : ∀ (e : Nat), s.ep = some e → s.epGen[e]? = some s.gen
  CallBoundary.Inv.reportedLe : s.reported ≤ s.gen
  CallBoundary.Inv.cmdsGen : ∀ (c : Cmd), c ∈ s.cmds → s.epGen[c.ep]? = some c.gen
  CallBoundary.Inv.callGen : ∀ (k : CallR), s.call = some k → k.gen ≤ s.gen ∧ (k.ended = false → s.reported ≤ k.gen)
