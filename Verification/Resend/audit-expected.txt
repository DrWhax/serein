def Resend.prune : S → Nat → S :=
fun s now =>
  have payloads := List.filter (fun x => decide (now ≤ x.ts + maxAge)) s.payloads;
  { payloads := payloads, rows := List.filter (fun r => payloads.any fun x => x.id == r.id) s.rows, next := s.next,
    now := s.now, stories := s.stories, lists := s.lists, counts := s.counts, delivered := s.delivered, out := s.out }
def Resend.payloadFor : S → Nat → Nat → S × Nat :=
fun s ts content =>
  match List.find? (fun p => p.ts == ts && p.content == content) s.payloads with
  | some p => (s, p.id)
  | none =>
    ({ payloads := s.payloads ++ [{ id := s.next, ts := ts, content := content }], rows := s.rows, next := s.next + 1,
        now := s.now, stories := s.stories, lists := s.lists, counts := s.counts, delivered := s.delivered,
        out := s.out },
      s.next)
def Resend.addRow : Nat → Nat → List Row → Nat → List Row :=
fun id recipient rows d =>
  if rows.contains { id := id, recipient := recipient, device := d } = true then rows
  else rows ++ [{ id := id, recipient := recipient, device := d }]
def Resend.record : S → Nat → Nat → Nat → List Nat → S :=
fun s ts content recipient devices =>
  have s1 := (payloadFor s ts content).fst;
  have id := (payloadFor s ts content).snd;
  prune
    { payloads := s1.payloads, rows := List.foldl (addRow id recipient) s1.rows devices, next := s1.next, now := s1.now,
      stories := s1.stories, lists := s1.lists, counts := s1.counts,
      delivered := s1.delivered ++ List.map (fun x => (ts, content, recipient, x)) devices, out := s1.out }
    s1.now
def Resend.receipt : S → Nat → Nat → Nat → S :=
fun s recipient device ts =>
  have atTs := List.filter (fun x => x.ts == ts) s.payloads;
  have rows :=
    List.filter (fun r => !(r.recipient == recipient && r.device == device && atTs.any fun x => x.id == r.id)) s.rows;
  { payloads := List.filter (fun p => !(p.ts == ts && !rows.any fun x => x.id == p.id)) s.payloads, rows := rows,
    next := s.next, now := s.now, stories := s.stories, lists := s.lists, counts := s.counts, delivered := s.delivered,
    out := s.out }
def Resend.find : S → Nat → Nat → Nat → Option Payload :=
fun s requester device ts =>
  List.find?
    (fun p =>
      p.ts == ts && decide (s.now ≤ p.ts + maxAge) &&
        s.rows.contains { id := p.id, recipient := requester, device := device })
    s.payloads
def Resend.listed : S → Nat → Nat → Bool :=
fun s requester ts =>
  s.stories.any fun x =>
    match x with
    | (t, l) => t == ts && s.lists.any fun L => L.id == l && !L.block && !L.deleted && L.members.contains requester
def Resend.answer : World → S → Nat → Nat → Nat → Nat → Answer :=
fun w s requester device ts answered =>
  if retryLimit ≤ answered then Answer.nothing
  else
    match find s requester device ts with
    | none => Answer.null
    | some p =>
      if (w.story.contains p.content && !listed s requester ts) = true then Answer.nothing else Answer.resend p
def Resend.countOf : S → Nat × Nat × Nat → Nat :=
fun s k => (Option.map (fun x => x.snd) (List.find? (fun x => x.fst == k) s.counts)).getD 0
def Resend.bump : List ((Nat × Nat × Nat) × Nat) → Nat × Nat × Nat → Nat → List ((Nat × Nat × Nat) × Nat) :=
fun counts k n => (k, n + 1) :: List.filter (fun x => x.fst != k) counts
def Resend.step : World → S → Ev → S :=
fun w s x =>
  match x with
  | Ev.send ts c r ds => record s ts c r ds
  | Ev.receipt r d ts => receipt s r d ts
  | Ev.tick n =>
    { payloads := s.payloads, rows := s.rows, next := s.next, now := s.now + n, stories := s.stories, lists := s.lists,
      counts := s.counts, delivered := s.delivered, out := s.out }
  | Ev.story ts l =>
    { payloads := s.payloads, rows := s.rows, next := s.next, now := s.now, stories := s.stories ++ [(ts, l)],
      lists := s.lists, counts := s.counts, delivered := s.delivered, out := s.out }
  | Ev.setList l =>
    { payloads := s.payloads, rows := s.rows, next := s.next, now := s.now, stories := s.stories,
      lists := l :: List.filter (fun x => x.id != l.id) s.lists, counts := s.counts, delivered := s.delivered,
      out := s.out }
  | Ev.retry a d ts =>
    have k := (a, d, ts);
    have n := countOf s k;
    have s :=
      { payloads := s.payloads, rows := s.rows, next := s.next, now := s.now, stories := s.stories, lists := s.lists,
        counts := bump s.counts k n, delivered := s.delivered, out := s.out };
    match answer w s a d ts n with
    | Answer.resend p =>
      { payloads := s.payloads, rows := s.rows, next := s.next, now := s.now, stories := s.stories, lists := s.lists,
        counts := s.counts, delivered := s.delivered,
        out := s.out ++ [{ requester := a, device := d, ts := ts, dest := a, carried := some (p.ts, p.content) }] }
    | Answer.null =>
      { payloads := s.payloads, rows := s.rows, next := s.next, now := s.now, stories := s.stories, lists := s.lists,
        counts := s.counts, delivered := s.delivered,
        out := s.out ++ [{ requester := a, device := d, ts := ts, dest := a, carried := none }] }
    | Answer.nothing => s

def Resend.Logged : S → Prop :=
fun s =>
  ∀ (r : Row), r ∈ s.rows → ∃ p, p ∈ s.payloads ∧ p.id = r.id ∧ (p.ts, p.content, r.recipient, r.device) ∈ s.delivered
def Resend.ResentDelivered : S → Prop :=
fun s =>
  ∀ (o : Out),
    o ∈ s.out →
      o.dest = o.requester ∧
        ∀ (c : Nat × Nat), o.carried = some c → c.fst = o.ts ∧ (c.fst, c.snd, o.requester, o.device) ∈ s.delivered
def Resend.outCount : S → Nat × Nat × Nat → Nat :=
fun s k => (List.filter (fun o => o.requester == k.fst && o.device == k.snd.fst && o.ts == k.snd.snd) s.out).length
def Resend.Bounded : S → Prop :=
fun s => ∀ (k : Nat × Nat × Nat), outCount s k ≤ retryLimit
