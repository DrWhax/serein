def Group.rotation : List (Nat × Nat) → List (Nat × Nat) → List Nat → Bool × List (Nat × Nat) :=
fun distributed current targets =>
  have reset := distributed.any fun d => !current.contains d && !targets.contains d.fst;
  (reset, if reset = true then current else List.filter (fun d => !distributed.contains d) current)
def Group.planOf : List Nat → List Nat → Bool → List Nat × List Nat :=
fun members rs update =>
  have rs := rs.eraseDups;
  (List.filter members.contains rs, if update = true then List.filter (fun x => !members.contains x) rs else [])
def Group.planJob : S → Nat → Job → S :=
fun s k j =>
  have p := planOf s.members j.rs j.update;
  have targets := List.filter (fun a => !(s.devices a).isEmpty) p.fst;
  have current := devs s targets;
  have r := rotation s.distributed current targets;
  have reset := r.fst || s.stale;
  have give := if reset = true then current else List.filter (fun x => !s.delivered.contains x) current;
  have base :=
    if reset = true then
      { members := s.members, devices := s.devices, epoch := s.epoch + 1, stale := false, distributed := [],
        jobs := s.jobs, holders := s.holders, out := s.out, delivered := [] }
    else s;
  { members := base.members, devices := base.devices, epoch := base.epoch, stale := base.stale,
    distributed := base.distributed ++ List.filter (fun x => !base.distributed.contains x) current,
    jobs :=
      base.jobs.set k
        { rs := j.rs, update := j.update, plan := some (targets, p.snd, (List.map (fun x => x.fst) give).eraseDups),
          generation := base.epoch, reserved := current },
    holders := base.holders, out := base.out, delivered := base.delivered }
def Group.stepCore : S → Ev → S :=
fun s x =>
  match x with
  | Ev.add a =>
    { members := setIn s.members a true, devices := s.devices, epoch := s.epoch, stale := s.stale,
      distributed := s.distributed, jobs := s.jobs, holders := s.holders, out := s.out, delivered := s.delivered }
  | Ev.remove a =>
    { members := setIn s.members a false, devices := s.devices, epoch := s.epoch, stale := s.stale,
      distributed := s.distributed, jobs := s.jobs, holders := s.holders, out := s.out, delivered := s.delivered }
  | Ev.refresh a ds =>
    { members := s.members, devices := upd s.devices a ds, epoch := s.epoch, stale := s.stale,
      distributed := s.distributed, jobs := s.jobs, holders := s.holders, out := s.out, delivered := s.delivered }
  | Ev.expire =>
    { members := s.members, devices := s.devices, epoch := s.epoch, stale := true, distributed := s.distributed,
      jobs := s.jobs, holders := s.holders, out := s.out, delivered := s.delivered }
  | Ev.queue rs u =>
    { members := s.members, devices := s.devices, epoch := s.epoch, stale := s.stale, distributed := s.distributed,
      jobs := s.jobs ++ [{ rs := rs, update := u, plan := none }], holders := s.holders, out := s.out,
      delivered := s.delivered }
  | Ev.plan k =>
    match s.jobs[k]? with
    | some j => planJob s k j
    | none => s
  | Ev.distribute k =>
    match s.jobs[k]? with
    | some j =>
      match j.plan with
      | some (targets, outside, a :: rest) =>
        if s.members.contains a = true then
          have ds := List.filter (fun x => x.fst == a) j.reserved;
          { members := s.members, devices := s.devices, epoch := s.epoch, stale := s.stale,
            distributed := s.distributed,
            jobs :=
              s.jobs.set k
                { rs := j.rs, update := j.update, plan := some (targets, outside, rest), generation := j.generation,
                  reserved := j.reserved },
            holders := s.holders ++ List.map (fun d => (d.fst, d.snd, j.generation)) ds, out := s.out ++ [(a, "key")],
            delivered :=
              if (j.generation == s.epoch) = true then s.delivered ++ List.filter (fun x => !s.delivered.contains x) ds
              else s.delivered }
        else unplan s k j
      | x => s
    | none => s
  | Ev.submit k =>
    match s.jobs[k]? with
    | some j =>
      match j.plan with
      | some (targets, outside, []) =>
        if
            (targets.all s.members.contains && j.generation == s.epoch && !s.stale &&
                s.distributed.all fun d => s.members.contains d.fst) =
              true then
          { members := s.members, devices := s.devices, epoch := s.epoch, stale := s.stale,
            distributed := s.distributed, jobs := s.jobs.eraseIdx k, holders := s.holders,
            out :=
              s.out ++ List.map (fun x => (x, if j.update = true then "update" else "content")) targets ++
                List.map (fun x => (x, "update")) outside,
            delivered := s.delivered }
        else unplan s k j
      | x => s
    | none => s

def Group.keysToMembers : S → Ev → S → Bool :=
fun s _e s' =>
  (newOut s s').all fun x =>
    match x with
    | (a, k) => k != "key" || s.members.contains a
def Group.contentToMembers : S → Ev → S → Bool :=
fun s _e s' =>
  (newOut s s').all fun x =>
    match x with
    | (a, k) => k != "content" || s.members.contains a
def Group.holdersRecorded : S → Ev → S → Bool :=
fun _s _e s' =>
  s'.holders.all fun x =>
    match x with
    | (a, d, g) => g != s'.epoch || s'.distributed.contains (a, d)
def Group.contentKeySafe : S → Ev → S → Bool :=
fun s e s' =>
  (!(newOut s s').any fun x => x.snd == "content") ||
    match e with
    | Ev.submit k =>
      match s.jobs[k]? with
      | some j => !s.stale && s.holders.all fun h => h.snd.snd != j.generation || s.members.contains h.fst
      | none => false
    | x => false
def Group.rotated : S → Ev → S → Bool :=
fun s e s' =>
  match e with
  | Ev.plan k =>
    match s.jobs[k]?, s'.jobs[k]? with
    | some val,
      some
        { rs := rs, update := update, plan := some (targets, fst, snd), generation := generation,
          reserved := reserved } =>
      s'.distributed.all fun d => targets.contains d.fst && s.members.contains d.fst
    | x, x_1 => true
  | x => true
