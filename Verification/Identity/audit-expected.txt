def Identity.step : S → Ev → S :=
fun s e =>
  stepCore
    { aci := s.aci, ourKey := s.ourKey, rows := s.rows, held := s.held, pending := s.pending, jobs := s.jobs,
      sent := s.sent, keyChanges := s.keyChanges, localNotices := s.localNotices, remoteNotices := s.remoteNotices,
      last := Res.none, shown := s.shown, grant := s.grant }
    e
def Identity.stepCore : S → Ev → S :=
fun s x =>
  match x with
  | Ev.view p => showSnap p s
  | Ev.confirm is =>
    { aci := s.aci, ourKey := s.ourKey, rows := s.rows, held := s.held,
      pending := s.pending ++ [List.filterMap (fun x => s.held[x]?) is], jobs := s.jobs, sent := s.sent,
      keyChanges := s.keyChanges, localNotices := s.localNotices, remoteNotices := s.remoteNotices, last := s.last,
      shown := s.shown, grant := s.grant }
  | Ev.commit k => commitAt k s
  | Ev.race k p kk => commitAt k (saveKey p kk s)
  | Ev.verify i b =>
    match s.held[i]? with
    | some x => setVer x b s
    | none => s
  | Ev.key p k => saveKey p k s
  | Ev.account a k =>
    { aci := a, ourKey := k, rows := s.rows, held := s.held, pending := s.pending, jobs := s.jobs, sent := s.sent,
      keyChanges := s.keyChanges, localNotices := s.localNotices, remoteNotices := s.remoteNotices, last := s.last,
      shown := s.shown, grant := s.grant }
  | Ev.window p =>
    match s.rows p with
    | some r =>
      { aci := s.aci, ourKey := s.ourKey,
        rows :=
          upd s.rows p
            (some
              { key := r.key, firstUse := r.firstUse, recent := false, verified := r.verified,
                approved := r.approved }),
        held := s.held, pending := s.pending, jobs := s.jobs, sent := s.sent, keyChanges := s.keyChanges,
        localNotices := s.localNotices, remoteNotices := s.remoteNotices, last := s.last, shown := s.shown,
        grant := s.grant }
    | none => s
  | Ev.storage p k v f => storage p k v f s
  | Ev.sync j =>
    match s.jobs[j]? with
    | some x =>
      { aci := s.aci, ourKey := s.ourKey, rows := s.rows, held := s.held, pending := s.pending,
        jobs := s.jobs.eraseIdx j, sent := s.sent ++ [x], keyChanges := s.keyChanges, localNotices := s.localNotices,
        remoteNotices := s.remoteNotices, last := s.last, shown := s.shown, grant := s.grant }
    | none => s

def Identity.saveKey : Nat → Nat → S → S :=
fun p k s =>
  match s.rows p with
  | none =>
    { aci := s.aci, ourKey := s.ourKey,
      rows :=
        upd s.rows p (some { key := k, firstUse := true, recent := true, verified := VS.dflt, approved := false }),
      held := s.held, pending := s.pending, jobs := s.jobs, sent := s.sent, keyChanges := s.keyChanges,
      localNotices := s.localNotices, remoteNotices := s.remoteNotices, last := s.last, shown := s.shown,
      grant := s.grant }
  | some r =>
    if (r.key != k) = true then
      have marked := r.verified == VS.verified || r.verified == VS.unverified;
      { aci := s.aci, ourKey := s.ourKey,
        rows :=
          upd s.rows p
            (some
              { key := k, firstUse := false, recent := true,
                verified := if marked = true then VS.unverified else VS.dflt, approved := false }),
        held := s.held, pending := s.pending, jobs := s.jobs, sent := s.sent, keyChanges := bump s.keyChanges p,
        localNotices := s.localNotices, remoteNotices := s.remoteNotices, last := s.last, shown := s.shown,
        grant := upd s.grant p none }
    else s
def Identity.valid : S → Snap → Bool :=
fun s x =>
  x.aci == s.aci && x.ourKey == s.ourKey &&
    match s.rows x.peer with
    | some r => r.key == x.key
    | none => false
def Identity.approveOne : S → Snap → S :=
fun s x =>
  match s.rows x.peer with
  | none => s
  | some r =>
    have s :=
      if (r.verified == VS.unverified) = true then
        { aci := s.aci, ourKey := s.ourKey,
          rows :=
            upd s.rows x.peer
              (some
                { key := r.key, firstUse := r.firstUse, recent := r.recent, verified := VS.dflt,
                  approved := r.approved }),
          held := s.held, pending := s.pending, jobs := s.jobs ++ [{ peer := x.peer, key := x.key, verified := false }],
          sent := s.sent, keyChanges := s.keyChanges, localNotices := bump s.localNotices x.peer,
          remoteNotices := s.remoteNotices, last := s.last, shown := s.shown, grant := s.grant }
      else s;
    match s.rows x.peer with
    | some r =>
      { aci := s.aci, ourKey := s.ourKey,
        rows :=
          upd s.rows x.peer
            (some
              { key := r.key, firstUse := r.firstUse, recent := r.recent, verified := r.verified, approved := true }),
        held := s.held, pending := s.pending, jobs := s.jobs, sent := s.sent, keyChanges := s.keyChanges,
        localNotices := s.localNotices, remoteNotices := s.remoteNotices, last := s.last, shown := s.shown,
        grant := upd s.grant x.peer (some (x.key, Src.user)) }
    | none => s
def Identity.approve : List Snap → S → S :=
fun xs s =>
  if xs.all (valid s) = true then
    have __src := List.foldl approveOne s xs;
    { aci := __src.aci, ourKey := __src.ourKey, rows := __src.rows, held := __src.held, pending := __src.pending,
      jobs := __src.jobs, sent := __src.sent, keyChanges := __src.keyChanges, localNotices := __src.localNotices,
      remoteNotices := __src.remoteNotices, last := Res.ok, shown := __src.shown, grant := __src.grant }
  else
    { aci := s.aci, ourKey := s.ourKey, rows := s.rows, held := s.held, pending := s.pending, jobs := s.jobs,
      sent := s.sent, keyChanges := s.keyChanges, localNotices := s.localNotices, remoteNotices := s.remoteNotices,
      last := Res.stale, shown := s.shown, grant := s.grant }
def Identity.setVer : Snap → Bool → S → S :=
fun x b s =>
  if (!valid s x) = true then
    { aci := s.aci, ourKey := s.ourKey, rows := s.rows, held := s.held, pending := s.pending, jobs := s.jobs,
      sent := s.sent, keyChanges := s.keyChanges, localNotices := s.localNotices, remoteNotices := s.remoteNotices,
      last := Res.stale, shown := s.shown, grant := s.grant }
  else
    match s.rows x.peer with
    | none =>
      { aci := s.aci, ourKey := s.ourKey, rows := s.rows, held := s.held, pending := s.pending, jobs := s.jobs,
        sent := s.sent, keyChanges := s.keyChanges, localNotices := s.localNotices, remoteNotices := s.remoteNotices,
        last := Res.ok, shown := s.shown, grant := s.grant }
    | some r =>
      have st := if b = true then VS.verified else VS.dflt;
      if (r.verified == st) = true then
        { aci := s.aci, ourKey := s.ourKey, rows := s.rows, held := s.held, pending := s.pending, jobs := s.jobs,
          sent := s.sent, keyChanges := s.keyChanges, localNotices := s.localNotices, remoteNotices := s.remoteNotices,
          last := Res.ok, shown := s.shown, grant := s.grant }
      else
        have r' :=
          if b = true then { key := r.key, firstUse := false, recent := r.recent, verified := st, approved := true }
          else { key := r.key, firstUse := r.firstUse, recent := r.recent, verified := st, approved := r.approved };
        { aci := s.aci, ourKey := s.ourKey, rows := upd s.rows x.peer (some r'), held := s.held, pending := s.pending,
          jobs := s.jobs ++ [{ peer := x.peer, key := x.key, verified := b }], sent := s.sent,
          keyChanges := s.keyChanges, localNotices := bump s.localNotices x.peer, remoteNotices := s.remoteNotices,
          last := Res.ok, shown := s.shown,
          grant := if b = true then upd s.grant x.peer (some (x.key, Src.user)) else s.grant }
def Identity.storage : Nat → Nat → VS → Bool → S → S :=
fun p k v first s =>
  have lo := s.rows p;
  have keysMatch :=
    match lo with
    | some r => r.key == k
    | none => false;
  if (first || lo.isNone || keysMatch) = true then
    have oldV :=
      match lo with
      | some r => r.verified
      | none => VS.dflt;
    have s :=
      if (lo.isSome && !keysMatch) = true then
        { aci := s.aci, ourKey := s.ourKey, rows := s.rows, held := s.held, pending := s.pending, jobs := s.jobs,
          sent := s.sent, keyChanges := bump s.keyChanges p, localNotices := s.localNotices,
          remoteNotices := s.remoteNotices, last := s.last, shown := s.shown, grant := s.grant }
      else s;
    have s :=
      if (!keysMatch || oldV != v) = true then
        { aci := s.aci, ourKey := s.ourKey,
          rows :=
            upd s.rows p (some { key := k, firstUse := lo.isNone, recent := true, verified := v, approved := true }),
          held := s.held, pending := s.pending, jobs := s.jobs, sent := s.sent, keyChanges := s.keyChanges,
          localNotices := s.localNotices, remoteNotices := s.remoteNotices, last := s.last, shown := s.shown,
          grant := upd s.grant p (some (k, Src.sync)) }
      else s;
    have became := v == VS.verified && (lo.isNone || oldV != VS.verified);
    have noLonger := v != VS.verified && lo.isSome && oldV == VS.verified;
    if (!(lo.isSome && !keysMatch) && (became || noLonger)) = true then
      { aci := s.aci, ourKey := s.ourKey, rows := s.rows, held := s.held, pending := s.pending, jobs := s.jobs,
        sent := s.sent, keyChanges := s.keyChanges, localNotices := s.localNotices,
        remoteNotices := bump s.remoteNotices p, last := s.last, shown := s.shown, grant := s.grant }
    else s
  else s
def Identity.commitOf : S → Ev → Option (S × List Snap) :=
fun s x =>
  match x with
  | Ev.commit k =>
    Option.map
      (fun xs =>
        ({ aci := s.aci, ourKey := s.ourKey, rows := s.rows, held := s.held, pending := s.pending.eraseIdx k,
            jobs := s.jobs, sent := s.sent, keyChanges := s.keyChanges, localNotices := s.localNotices,
            remoteNotices := s.remoteNotices, last := Res.none, shown := s.shown, grant := s.grant },
          xs))
      s.pending[k]?
  | Ev.race k p kk =>
    have t :=
      saveKey p kk
        { aci := s.aci, ourKey := s.ourKey, rows := s.rows, held := s.held, pending := s.pending, jobs := s.jobs,
          sent := s.sent, keyChanges := s.keyChanges, localNotices := s.localNotices, remoteNotices := s.remoteNotices,
          last := Res.none, shown := s.shown, grant := s.grant };
    Option.map
      (fun xs =>
        ({ aci := t.aci, ourKey := t.ourKey, rows := t.rows, held := t.held, pending := t.pending.eraseIdx k,
            jobs := t.jobs, sent := t.sent, keyChanges := t.keyChanges, localNotices := t.localNotices,
            remoteNotices := t.remoteNotices, last := t.last, shown := t.shown, grant := t.grant },
          xs))
      t.pending[k]?
  | x => none
def Identity.trust : List Nat → S → List (Option Row × Option (Nat × Src) × Nat × Nat × Nat) :=
fun ps s => List.map (fun p => (s.rows p, s.grant p, s.keyChanges p, s.localNotices p, s.remoteNotices p)) ps
def Identity.staleNoChange : List Nat → S → Ev → S → Bool :=
fun ps s e s' =>
  match commitOf s e with
  | some (t, xs) => xs.all (valid t) || trust ps s' == trust ps t && s'.jobs == t.jobs
  | none =>
    match e with
    | Ev.verify i b =>
      match s.held[i]? with
      | some x => valid s x || trust ps s' == trust ps s && s'.jobs == s.jobs
      | none => true
    | x => true
def Identity.targets : List Nat → S → Ev → S → Bool :=
fun ps s e s' =>
  match commitOf s e with
  | some (t, xs) =>
    (ps.all fun p =>
        t.rows p == s'.rows p ||
          match t.rows p, s'.rows p with
          | some r, some r' => r.key == r'.key && xs.any fun x => x.peer == p && x.key == r.key
          | x, x_1 => false) &&
      (!xs.all (valid t) ||
        xs.all fun x =>
          match s'.rows x.peer with
          | some r' => r'.key == x.key && r'.approved && r'.verified != VS.unverified
          | none => false)
  | none =>
    match e with
    | Ev.verify i b =>
      match s.held[i]? with
      | some x =>
        ps.all fun p =>
          s.rows p == s'.rows p ||
            p == x.peer &&
              match s.rows p, s'.rows p with
              | some r, some r' =>
                r.key == r'.key && r.key == x.key && (!b || r'.verified == VS.verified && r'.approved)
              | x, x_1 => false
      | none => true
    | x => true
def Identity.grantShown : List Nat → S → Ev → S → Bool :=
fun ps _s _e s' =>
  ps.all fun p =>
    match s'.rows p with
    | some r =>
      !(r.approved || r.verified == VS.verified) ||
        match s'.grant p with
        | some (k, src) => k == r.key && (src != Src.user || s'.shown.any fun x => x.peer == p && x.key == k)
        | none => false
    | none => true
def Identity.syncShown : S → Ev → S → Bool :=
fun _s _e s' => (s'.jobs ++ s'.sent).all fun j => s'.shown.any fun x => x.peer == j.peer && x.key == j.key
def Identity.syncExact : S → Ev → S → Bool :=
fun s e s' =>
  match e with
  | Ev.sync j =>
    match s.jobs[j]? with
    | some x => s'.sent == s.sent ++ [x]
    | none => s'.sent == s.sent
  | x => s'.sent == s.sent
def Identity.keyChangeRevokes : S → Ev → S → Bool :=
fun s e s' =>
  match e with
  | Ev.key p k =>
    match s.rows p with
    | some r =>
      r.key == k ||
        (match s'.rows p with
          | some r' => !r'.approved && r'.verified != VS.verified && s'.grant p == none
          | none => false) &&
          s'.held.all fun x => x.peer != p || x.key != r.key || !valid s' x
    | none => true
  | Ev.account a k =>
    a == s.aci && k == s.ourKey || s'.held.all fun x => !(x.aci == s.aci && x.ourKey == s.ourKey) || !valid s' x
  | x => true
def Identity.storageNoSubstitute : S → Ev → S → Bool :=
fun s e s' =>
  match e with
  | Ev.storage p k v false =>
    match s.rows p with
    | some r => r.key == k || s'.rows p == s.rows p && s'.grant p == s.grant p
    | none => true
  | x => true

def Identity.GrantOk : S → Prop :=
fun s =>
  ∀ (p : Nat) (r : Row),
    s.rows p = some r →
      r.approved = true ∨ r.verified = VS.verified →
        ∃ src, s.grant p = some (r.key, src) ∧ (src = Src.user → ∃ x, x ∈ s.shown ∧ x.peer = p ∧ x.key = r.key)
